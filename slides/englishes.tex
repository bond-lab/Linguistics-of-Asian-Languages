\section{World Englishes}

\begin{frame}{World Englishes}
  \Large
  \begin{itemize}
  \item The \txx{varieties} of English used around the world
  \item English as a global \txx{lingua franca}, and English made \txx{local}
  \item British, Indian, Chinese, Nigerian English \ldots
  \end{itemize}
\end{frame}

\begin{frame}{Kachru's Three Circles of English}
\centering
\begin{tikzpicture}[scale=0.72, every node/.style={transform shape}]
    \fill[fill=pink!20] (0,0) circle [radius=4.5];
    \fill[fill=pink!50] (-1,0) circle [radius=3];

\end{tikzpicture}

\bigskip
\small After \citet{Kachru:1985}
\end{frame}

\begin{frame}{The Three Circles}
  \normalsize
  \begin{tabular}{lll}
    \toprule
    & English is \ldots & for example \\
    \midrule
    \txx{inner}     & the main language; the ``standard'' & UK, USA, Australia \\
    \txx{outer}     & a second, official language         & India, Nigeria, Singapore \\
    \txx{expanding} & a foreign language                  & China, Japan, Brazil \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  \begin{itemize}
  \item Outer: a colonial legacy, nativized varieties, much multilingualism ---
    and more speakers than the inner circle
  \item Expanding: no colonial legacy and no official role, but the
    \emph{most} speakers
  \item Singapore fits almost every inner-circle criterion except origin
  \end{itemize}
\end{frame}

\end{frame}

\begin{frame}{Quirk vs.\ Kachru}
  \large
  \begin{tabular}{lll}
    \toprule
    & \citet{Quirk:1985we}: uniformity & \citet{Kachru:1985}: pluralism \\
    \midrule
    standard   & one, for intelligibility     & many: diversity is real \\
    teaching   & inner-circle norms           & validate local varieties \\
    ownership  & the inner circle             & all its users \\
    ``errors'' & a risk of misunderstanding   & a deficit model to reject \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  ``We need one standard English for international communication'' vs
  ``English belongs to all its users, not just its historical native speakers''
\end{frame}

\begin{frame}{Four False Assumptions (after Kachru)}
  \normalsize
  \begin{tabular}{ll}
    \toprule
    the assumption & the counterpoint \\
    \midrule
    there is one homogeneous standard     & inner-circle English varies too \\
    outer-circle English is deficient     & it is a legitimate local adaptation \\
    teach inner-circle norms only         & teach the local reality \\
    inner-circle norms ensure intelligibility & intelligibility is mutual, \\
                                          & \quad and depends on context \\
    \bottomrule
  \end{tabular}
  \vfill
  \small A paraphrase of the argument in \citet{Kachru:1992}
\end{frame}

\begin{frame}{Innovation, Deviation or Mistake?}
  \large
  \begin{tabular}{ll}
    \toprule
    \txx{innovation} & creative use, often denied to non-inner speakers \\
                     & \quad \eng{timepass}, \eng{skinship}, \eng{add oil} \\
    \txx{deviation}  & differs from some other variety's norm \\
                     & \quad \eng{He is knowing the answer.} \\
    \txx{mistake}    & a learner's error, not a community's norm \\
                     & \quad \eng{He knowed the answer.} \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  \citet{Kachru:1992}.  Different varieties are held to different standards.
  But careful: regularized past tenses like \eng{knowed} \emph{are} systematic
  in some Appalachian and African American speech --- so whose mistake is it?
\end{frame}

\section{Variation in the Inner Circle}

\begin{frame}{Three ``Standards'': BrE, AmE, AusE}
  \Large
  \begin{tabular}{ll}
    \toprule
    extended meanings & \eng{corn}, \eng{robin} \\
    new words         & \eng{buttle} ``to act as a butler'' \\
    borrowings        & \eng{moccasin}, \eng{squash}, \eng{toboggan} \\
    later technology  & \eng{windshield} (US) vs \eng{windscreen} (UK) \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Vocabulary is the most noticeable difference between North American and
  British English; pronunciation differs too, and each region has its own
  dialects
\end{frame}

\begin{frame}{Australian English Loves a Clipping}
  \large
  \begin{tabular}{llll}
    \toprule
    \eng{barbie} & barbecue    & \eng{Snowy}  & someone with white hair \\
    \eng{uni}    & university  & \eng{Bluey}  & a red-head \\
    \eng{sanger} & sandwich    & \eng{Bondie} & Bond (me) \\
    \eng{relly}  & relative    & \eng{chuck a U-ey} & make a U-turn \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Also loans from Aboriginal languages: \eng{kangaroo}, \eng{boomerang}
\end{frame}

\begin{frame}{American and British Grammar Differ}
  \large
  \begin{tabular}{lll}
    \toprule
    & US & UK \\
    \midrule
    verbs      & \eng{He already ate.}             & \eng{He has already eaten.} \\
    nouns      & \eng{The team is winning.}        & \eng{The team are winning.} \\
    adverbs    & \eng{He runs real fast.}          & \eng{He runs really fast.} \\
    adjectives & \eng{How big of a room is it?}    & \eng{How big a room is it?} \\
    prepositions & \eng{I went on the weekend.}    & \eng{I went at the weekend.} \\
    \bottomrule
  \end{tabular}
  \vfill
  \small \citet{Trudgill:Hannah:2002}
\end{frame}

\begin{frame}{Prestige Hierarchies Everywhere}
  \normalsize
  \begin{tabular}{ll}
    \toprule
    Australian English \eng{Strine} & once ``broad'', ``lazy''; now a national standard \\
    AAVE \eng{Ebonics}             & rule-governed: copula absence, habitual \eng{be} \\
    Appalachian, Southern US       & a-prefixing \eng{He's a-running}, \\
                                   & \quad double modals \eng{He might could do it} \\
    Scouse, Geordie, Glaswegian    & local prestige, little institutional prestige \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  The inner circle is not a uniform standard: all Englishes have prestige
  hierarchies, with \txx{overt} and \txx{covert} prestige
  \citep{Burridge:Mulder:1999,Rickford:1999,LippiGreen:2012}
\end{frame}

\section{Speakers and Labels}

\begin{frame}{What's Wrong with ``Native Speaker''?}
  \Large
  \begin{itemize}
  \item Treats \txx{monolingualism} as normal
  \item Overrates the order of acquisition
  \item Makes Anglo speakers the reference point, and power one-way
  \end{itemize}
  \vfill
  \normalsize
  It also encourages a simple view of ``errors''.  What does it say about
  \emph{you} as speakers of English?
\end{frame}

\begin{frame}{Alternatives}
  \large
  \begin{tabular}{ll}
    \toprule
    \multicolumn{2}{l}{\citet{Rampton:1990}: \txx{expertise} instead} \\
    \midrule
    + & learned, not innate; relative, partial, contestable \\
    $-$ & ``non-expert'' is still a value judgement \\
    \midrule
    \multicolumn{2}{l}{\citet{Jenkins:2000}: MES, BES, NBES} \\
    \midrule
    MES  & monolingual English speaker \\
    BES  & bilingual English speaker \\
    NBES & non-bilingual English speaker \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  The point: speakers' whole \txx{repertoire}, not native vs non-native
\end{frame}

\begin{frame}{Names Are Not Neutral}
  \normalsize
  \begin{tabular}{lll}
    \toprule
    variety & label & connotation \\
    \midrule
    Indian English         & \eng{Hinglish}   & code-mixing, youth culture; can stereotype \\
    Singapore English      & \eng{Singlish}   & officially a problem; locally, solidarity \\
    Hong Kong English      & \eng{Chinglish}  & mocking, about ``errors'' \\
    Japanese English       & \eng{Engrish}    & from outside; mocking, racist \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Linguists say \emph{Colloquial Singapore English}, \emph{Japanese English},
  \emph{wasei-eigo}.  Labels index attitudes, power and ownership
\end{frame}

\begin{frame}{Activity: How Would You Label Yourselves?}
  \large
  \begin{itemize}
  \item NS / NNS?  MES / BES / NBES?  Something else?
  \item How do the labels affect
    \begin{itemize}
    \item how confident you feel speaking English?
    \item how you judge other people's English?
    \end{itemize}
  \item Who gets to name a variety: the state, teachers, speakers, outsiders?
    And how does the name affect its \emph{prestige} and \emph{legitimacy}?
  \end{itemize}
\end{frame}

\section{Asian Englishes}

\begin{frame}{Schneider's Dynamic Model}
  \normalsize
  \begin{tabular}{rll}
    \toprule
    1 & foundation                  & English arrives \\
    2 & exonormative stabilization  & the norm comes from outside \\
    3 & nativization                & local forms spread: \alert{India, HK, Philippines} \\
    4 & endonormative stabilization & the local norm is accepted: \alert{Singapore} \\
    5 & differentiation             & new internal dialects \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Unlike Kachru's static circles, a path every postcolonial English follows,
  shaped by settlers' and locals' identities.  It explains why Singapore
  looks inner-circle \citep{schneider:2007}
\end{frame}

\begin{frame}{What Asian Englishes Share}
  \Large
  \begin{tabular}{ll}
    \toprule
    phonology & local accents and stress: Indian retroflexes \\
    syntax    & local influence: Singapore drops articles \\
    lexicon   & borrowings: Malaysian \eng[seal, stamp]{chop} \\
    use       & extensive code-switching \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Codification brings acceptance, prestige and a classroom model --- but
  local varieties are still seen as ``interlanguages'', measured against
  native-speaker competence
\end{frame}

\section{Indian English}

\begin{frame}{Indian English}
  \large
  \begin{tabular}{ll}
    \toprule
    pronunciation & variably rhotic (especially in the south); different \\
                  & \quad vowels in \eng{bat}, \eng{bad}, \eng{bed} \\
    vocabulary    & \eng[reschedule earlier]{prepone}, \eng[warehouse]{godown} \\
    grammar       & stative progressive: \eng{He is knowing the answer.} \\
                  & \eng{doubt} ``question'': \eng{I have a doubt in maths.} \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Shaped by India's many languages; used in government, education, business
  and media, and a bridge in a multilingual society
\end{frame}

\begin{frame}{Indian English Vocabulary}
  \Large
  \begin{tabular}{ll}
    \toprule
    \multicolumn{2}{l}{\txx{semantic shifts}} \\
    \midrule
    \eng{pass out}        & graduate \\
    \eng{out of station}  & out of town \\
    \eng{revert}          & reply (also Singapore, Malaysia) \\
    \midrule
    \multicolumn{2}{l}{\txx{local coinages}} \\
    \midrule
    \eng{hill station}    & mountain resort \\
    \eng{timepass}        & leisure activity \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Some Indian words have gone into general English: \eng{bungalow} (Hindi
  \mtciteform{bangla}), \eng{jungle} (\mtciteform{jangal})
\end{frame}

\begin{frame}{Indian English in Use}
  \Large
  \begin{itemize}
  \item \eng{Can you \alert{prepone} the meeting to tomorrow?}
  \item \eng{I \alert{passed out} of college in 2020.}
  \item \eng{He is \alert{having a doubt} in mathematics.}
  \item \eng{She went to buy vegetables \alert{only}.}
  \item \eng{Let us go for a walk in the evening, \alert{no}?}
  \end{itemize}
\end{frame}

\section{Singapore English}

\begin{frame}{Singlish}
  \Large
  \begin{itemize}
  \item The colloquial English of Singapore
  \item English, with Malay, Tamil, Hokkien, Cantonese \ldots
  \item Not officially endorsed --- but central to Singaporean identity
  \end{itemize}
  \vfill
  \normalsize
  Despite the government's \emph{Speak Good English} campaigns, it thrives in
  media, humour and everyday talk
\end{frame}

\begin{frame}{Singlish Vocabulary}
  \large
  \begin{tabular}{lll}
    \toprule
    \lex{kiasu}   & \eng{He is so kiasu.}          & afraid to lose out \\
    \lex{shiok}   & \eng{This food is so shiok!}   & delicious, enjoyable \\
    \lex{blur}    & \eng{Why are you so blur?}     & confused \\
    \lex{paiseh}  & \eng{So paiseh to ask!}        & embarrassed \\
    \lex{ang moh} & \eng{The ang moh loves laksa.} & a Caucasian (``red hair'') \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Try our \href{https://singdict.github.io/}{Singlish Dictionary}!
\end{frame}

\begin{frame}{Singlish Grammar}
  \Large
  \begin{tabular}{ll}
    \toprule
    aspect by adverb & \eng{He go \alert{already}.} ``He has gone.'' \\
    particles        & \eng{Don't worry \alert{lah}!} emphasis, assurance \\
                     & \eng{You want coffee, \alert{ah}?} questioning \\
                     & \eng{Very expensive, \alert{leh}.} softening, tentative \\

    \bottomrule
  \end{tabular}
\end{frame}

\section{Hong Kong English}

\begin{frame}{Hong Kong English}
  \large
  \begin{tabular}{ll}
    \toprule
    pronunciation & Cantonese tone shapes stress; \\
                  & \quad /n/ and /l/ merge (\eng{night}, \eng{light}), as in Cantonese \\
                  & \quad \zht{你} \yue{néih} $\to$ \yue{léih} \\
    grammar       & articles and prepositions dropped: \eng{I go market.} \\
    vocabulary    & \eng[drink tea]{yum cha}, \eng[roast pork]{char siu} \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  From British rule (1842--1997); still important in education, business and
  law.  Also: \eng{th} becomes [f] or [t], final clusters are simplified
  \citep[see][ch.~3]{Setter:Wong:Chan:2010}
\end{frame}

\begin{frame}{Hong Kong English Vocabulary}
  \Large
  \begin{tabular}{ll}
    \toprule
    \lex{dai pai dong}   & open-air food stall \\
    \lex{si fu}          & master, skilled worker \\
    \lex{cha chaan teng} & Hong Kong-style cafe \\
    \midrule
    \lex{add oil}        & keep going! good luck! \\
    \lex{long time no see} & perhaps \zht{好耐冇見} \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  \eng{He very smart, la.}  \eng{I no understand.}  \eng{I go yum cha with
  family tomorrow.}
\end{frame}

\section{Philippine English}

\begin{frame}{Philippine English: The American One}
  \normalsize
  \begin{tabular}{ll}
    \toprule
    history       & American rule 1898--1946; the \emph{Thomasite} teachers from 1901 \\
    pronunciation & rhotic, from American English; \\
                  & \quad /f/ often [p]: \eng{Pilipino} \\
    vocabulary    & \eng{comfort room} ``toilet'', \eng{carnap} ``steal a car'', \\
                  & \quad \eng{open the light} ``switch it on'' \\
    use           & \txx{Taglish}: mixing Tagalog and English \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  About 70 million speakers, the second largest in Asia --- and the only
  major Asian English from American, not British, colonial rule
\end{frame}

\section{The Future}

\begin{frame}{English as a Lingua Franca}
  \large
  \begin{tabular}{ll}
    \toprule
    \multicolumn{2}{l}{\citet{Jenkins:2000}'s \txx{Lingua Franca Core}: what intelligibility needs} \\
    \midrule
    core     & consonants (not \eng{th}); aspirated /p t k/; \\
             & \quad initial clusters; vowel length; nuclear stress \\
    not core & \eng{th}; exact vowel quality; weak forms; word stress \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  When most English is between non-native speakers, the target is mutual
  intelligibility, not a native accent --- one answer to ``how do we balance
  intelligibility and diversity?''
\end{frame}

\begin{frame}{Inner-Circle Variation Explains Asian Englishes}
  \normalsize
  \begin{tabular}{ll}
    \toprule
    covert prestige    & Strine, AAVE, Singlish: in-group solidarity \\
    standard ideology  & used against AAVE, UK dialects, Indian English \\
    error or innovation & AAVE rules long called errors; Asian features too \\
    ownership          & AAVE in schools $\approx$ teaching Indian English, Singlish \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Asian varieties are not special cases: \emph{all Englishes are dialects}
  \citep{Milroy:Milroy:1999}
\end{frame}

\begin{frame}{The Future of Asian Englishes}
  \Large
  \begin{itemize}
  \item Growing \txx{prestige}, in schools and online
  \item Will codification make \txx{new standards}?
  \item How do we balance \txx{intelligibility} and \txx{diversity}?
  \end{itemize}
  \vfill
  \normalsize
  How established a variety is depends on its size and its prestige
\end{frame}

\begin{frame}{Activity: Innovation, Deviation, or Mistake?}
  \large
  In pairs: innovation (accepted in some community), deviation (different
  from a chosen standard), or mistake (accepted nowhere)?
  \begin{exe}
    \ex \eng{Can you prepone the meeting to tomorrow?}
    \ex \eng{Yesterday I go market, very crowded, lah.}
    \ex \eng{He is knowing the answer.}
    \ex \eng{She is 22 years.}
    \ex \eng{This food is damn shiok!}
    \ex \eng{They knowed the answer.}
    \ex \eng{I need to control my homework.}
    \ex \eng{I beez in the trap.}
  \end{exe}
  \textbf{Whose norms are you using when you decide?}
\end{frame}
