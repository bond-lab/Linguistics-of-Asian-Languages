\section{Proverbs and Elaborate Expressions}

\begin{frame}{Proverbs Speak for You}
  \Large
  \begin{itemize}
  \item Central to communication, especially in \txx{Malay}
  \item Express social norms and values
  \item \txx{Depersonalize} advice: not my opinion, traditional wisdom
  \end{itemize}
\end{frame}

\begin{frame}{Malay Proverbs}
  \begin{exe}
    \ex
    \gll Ada hujan, ada panas, ada hari boleh balas. \\
         exist rain exist heat exist day can repay \\
    \trans ``Every deed, good or bad, will be repaid in kind.'' [zsm]
    \ex
    \gll Sedikit-sedikit, lama-lama jadi bukit. \\
         little-little long-long become hill \\
    \trans ``Little by little, over time, it becomes a hill.'' [zsm]
  \end{exe}
  \vfill
  \large
  Small, patient actions bring big results
\end{frame}

\begin{frame}{The Pantun: A Quatrain in Two Halves}
  \normalsize
  \begin{tabular}{l}
    \mtplain{Pisang emas bawa belayar,} \\
    \mtplain{\quad Masak sebiji di atas peti;} \\
    \mtplain{Hutang emas boleh dibayar,} \\
    \mtplain{\quad Hutang budi dibawa mati.} \\
  \end{tabular}
  \hfill
  \begin{tabular}{l}
    The golden plantain carried on a voyage; \\
    \quad One ripens on top of the chest. \\
    A debt of gold can be repaid; \\
    \quad A debt of kindness lasts until death. \\
  \end{tabular}
  \vfill
  \normalsize
  The first two lines set an image; the last two carry the meaning, linked
  by rhyme and indirect association.  See
  \href{https://ir.uitm.edu.my/id/eprint/63499/1/63499.pdf}{\textit{Connecting
  the dots: The case of foreshadow and meaning in pantun}}, Nazima Versay
  Kudus and Melati Desa (2022)
\end{frame}

\begin{frame}{Elaborate Expressions: Four Syllables, ABAC}
  \large
  \begin{tabular}{lll}
    \toprule
    & literally & means \\
    \midrule
    \ths{หูป่าตาเถื่อน} & ear-forest-eye-wild & ignorant \\
    \quad\tha{hǔu-pàa-taa-thɯ̀an} & & \\[1ex]
    \ths{พูดอย่างทำอย่าง} & speak-kind-do-kind & say one thing, \\
    \quad\tha{phûut-yàaŋ-tham-yàaŋ} & & do another \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Four-syllable compounds (ABAC, ABCB) are common across Thai, Lao,
  Vietnamese and Chinese, and a mark of fluent, vivid speech
\end{frame}

\section{Chengyu}

\begin{frame}{Chengyu \zhs{成语}: Set Phrases from the Classics}
  \Large
  \begin{itemize}
  \item Typically \txx{four characters}, from history or classical literature
  \item Partly or wholly \txx{non-compositional}
  \item Work as single words, but more vivid
  \end{itemize}
  \vfill
  \normalsize
  There is no definitive list: lexicons range from a few hundred to 50,000
  or more, and there are many excellent (paper) dictionaries
\end{frame}

\begin{frame}{Some Chengyu}
  \hfill \includegraphics{pics/shouzhudaitu.jpg}
  \vspace{-3cm}
  \begin{exe}
    \ex\label{s:rabbit}
    \glll 守 株 待 兔 \\
          shǒu zhū dài tù \\
          guard {tree stump} wait rabbit \\
    \trans ``to expect fortune without any effort'' [cmn]
    \ex\label{s:divide}
    \glll 分 崩 离 析 \\
          fēn bēng lí xī \\
          divide rupture leave {split apart} \\
    \trans ``to completely fall apart'' [cmn]
  \end{exe}
  \small Image from \href{https://www.dimsumwarriors.com/chengyu/守株待兔-shouzhudaitu/}{Yumcha Studios}
\end{frame}

\begin{frame}{Many Come from Stories}

  \end{tabular}
  \vfill

\begin{frame}{Chengyu Travel: Japanese and Korean}
  \Large
  \begin{tabular}{lll}
    \toprule
    Mandarin & \zhs{四面楚歌} & \cmn{sìmiàn chǔgē} \\
    Japanese & \jps{四面楚歌} & \jpn{shimen soka} \\
    Korean   & \kos{사면초가}  & \kor{samyeon choga} \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  \zhs{四面楚歌} ``songs of Chu on four sides'': surrounded by enemies.
  Used most in Chinese, then Korean, then Japanese.  Not all are Chinese:
  Japanese \jps{一石二鳥} \jpn{isseki nichō} ``one stone, two birds'' is said
  to translate the English proverb
\end{frame}

\begin{frame}{Not So Different from Western Proverbs}
  \Large
  \begin{tabular}{ll}
    \zhs{画蛇添足} \cmn{huà shé tiān zú} & ``draw legs on a snake'' \\
    & = add something superfluous \\[1ex]
    \eng{sour grapes}, \eng{lipstick on a pig} & also need interpreting \\
  \end{tabular}
  \vfill
  \normalsize
  Western proverbs also come from classic works (Aesop, the Bible, Homer),
  and using them well is a sign of erudition.  The fixed four-character form
  makes chengyu stand out more from ordinary text
\end{frame}

\section{Politeness and Speech Styles}

\begin{frame}{Politeness Protects Face}
  \large
  \begin{tabular}{ll}
    \toprule
    \txx{positive face} & the wish to be liked and approved of \\
    \txx{negative face} & the wish not to be imposed on \\
    \bottomrule
  \end{tabular}
  \vfill
  Requests, refusals and criticism are \txx{face-threatening acts}; speakers
  soften them:
  \vfill
  \begin{tabular}{ll}
    \toprule
    positive politeness & build closeness: Korean \kor{banmal}, nicknames \\
    negative politeness & show deference: honorifics, Javanese \mtciteform{krama} \\
    off record          & hint: Malay ``Are you hungry?'' \\
    don't do the act    & settle it first: Japanese \jpn{nemawashi} \\
    \bottomrule
  \end{tabular}
  \vfill
  \small \citet{brown:levinson:1987}
\end{frame}

\begin{frame}{Speech Styles}
  \Large
  \begin{itemize}
  \item Many languages here have distinct \txx{speech styles} (registers)
  \item They signal respect, distance and familiarity
  \item Marked in pronunciation, vocabulary, particles, morphology, syntax
  \end{itemize}
  \vfill
  \normalsize
  Most elaborate and codified in Javanese and Korean
\end{frame}

\begin{frame}{Thai: Polite Particles and Styles}
  \Large
  \begin{tabular}{lll}
    \toprule
    male speaker   & \ths{ครับ} \tha{khráp} & \\
    female speaker & \ths{ค่ะ} \tha{khâ}    & \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  \begin{itemize}
  \item Styles: public, consultative, personal (informal)
  \item Particles end a sentence or stand alone; consultative style also
    wants careful pronunciation, elevated words, and no abrupt or negative
    expressions
  \end{itemize}
\end{frame}

\begin{frame}{Korean: Speech Levels in the Verb Ending}
  \Large
  \begin{tabular}{lll}
    \toprule
    \multicolumn{3}{l}{``It is raining''} \\
    \midrule

    \bottomrule
  \end{tabular}
  \vfill
  \normalsize

\begin{frame}{Korean: Everyone Notices the Level}
  \Large
  \begin{tabular}{l}

    \quad ``We are on \kor{hage} terms with each other.'' \\[2ex]

    \quad ``Who do you think you're speaking \kor{banmal} to?'' \\
  \end{tabular}
  \vfill
  \normalsize

\begin{frame}{Korean: Honouring the Person Talked About}
  \Large
  \begin{tabular}{llll}
    \toprule
    & plain & & honorific \\
    \midrule
    go        & \kos{가다} \kor{gada}   & subject \kos{-시-} & \kos{가시다} \kor{gasida} \\

    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Speech levels honour the \txx{addressee}; these honour the \txx{referent}

\end{frame}

\begin{frame}{Speech Styles Compared}
  \Large
  \begin{itemize}
  \item Javanese, Sasak, Korean: \txx{structured levels}; Thai: royal vocabulary
  \item They encode \txx{hierarchy}: age, rank, social distance
  \item Common in stratified societies with fixed roles
  \end{itemize}
  \vfill
  \normalsize
  Shaped by Confucian, Buddhist and local values.  Knowing the styles is part
  of communicative competence
\end{frame}

\section{Japanese Honorifics}

\begin{frame}{Three Kinds of Japanese Honorific}
  \large
  \begin{tabular}{lll}
    \toprule
    \jpn{teineigo} & polite   & towards the \txx{addressee}: \jpn{-masu} \\
    \jpn{sonkeigo} & honouring & raises the \txx{referent} (usually the subject) \\
    \jpn{kenjōgo}  & humbling & lowers oneself or one's in-group \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  They show respect, social distance and group membership, and are part of
  everyday Japanese
\end{frame}

\begin{frame}{Honouring and Humbling Verbs}
  \Large
  \begin{tabular}{llll}
    \toprule
    & neutral & honouring & humbling \\
    \midrule
    say  & \jpn{iu}     & \jpn{ossharu}         & \jpn{mōsu} \\
    go   & \jpn{iku}    & \jpn{irassharu}       & \jpn{mairu} \\
    do   & \jpn{suru}   & \jpn{nasaru}          & \jpn{itasu} \\
    eat  & \jpn{taberu} & \jpn{meshiagaru}      & \jpn{itadaku} \\
    stay & \jpn{tomaru} & \jpn{o-tomari ni naru} & \jpn{o-tomari suru} \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Regular verbs: honouring \jpn{o-}stem \jpn{ni naru}, humbling \jpn{o-}stem
  \jpn{suru}.  Politeness is separate: \jpn{iu} vs \jpn{iimasu}.  Kansai
  (especially Kyoto) has a lighter honorific \jpn{V-haru}: \jpn{tabe-haru}
\end{frame}

\begin{frame}{In-Group and Out-Group: \jpn{uchi} and \jpn{soto}}
  \begin{exe}
    \ex
    \gll Chichi wa genki de ori-mas-u. \\
         father \textsc{top} healthy \textsc{cop} be.\textsc{hum}-\textsc{pol}-\textsc{npst} \\
    \trans ``(My) father is well.'' [jpn]
    \ex
    \gll O-tō-san wa o-genki des-u ka. \\
         \textsc{hon}-father-\textsc{hon} \textsc{top} \textsc{hon}-health \textsc{cop}.\textsc{pol}-\textsc{npst} \textsc{q} \\
    \trans ``Is (your) father well?'' [jpn]
  \end{exe}
  \vfill
  \large
  Humble for your own family, school or company; honouring for outsiders
\end{frame}

\begin{frame}{Humbling: Doing Something for Someone Higher}
  \begin{exe}
    \ex
    \gll Sakai-san ga Suzuki-san ni chizu o \ul{o-kaki} \ul{shi}-mashi-ta. \\
         Sakai-\textsc{hon} \textsc{nom} Suzuki-\textsc{hon} \textsc{dat} map \textsc{acc} \textsc{hon}-draw do-\textsc{pol}-\textsc{pst} \\
    \trans ``Mr Sakai drew a map for Mr Suzuki (humbly).'' [jpn]
  \end{exe}
  \vfill
  \large
  Women are expected to use more elaborate honorifics than men; the
  difference shrinks at the highest levels
\end{frame}

\begin{frame}{Giving and Receiving}
  \Large

\begin{frame}{Names and Titles}
  \large
  \begin{tabular}{lll}
    \toprule
    \jps{さん} \jpn{san}     & Mr / Ms        & adults of equal status \\
    \jps{様} \jpn{sama}      & Sir / Madam    & higher status: guests, customers, deities \\
    \jps{君} \jpn{kun}       & ---            & juniors, boys, male friends \\
    \jps{ちゃん} \jpn{chan}  & little \ldots  & small children, cute things, close friends \\
    \jps{たん} \jpn{tan}     & widdle \ldots  & babies, moe characters \\
    \jps{先輩} \jpn{senpai}  & ---            & a senior colleague or classmate \\
    \jps{先生} \jpn{sensei}  & Dr / Prof      & teachers, doctors, lawyers, authors \\
    \jps{行} \jpn{yuki}      & ---            & your own name, on a reply envelope \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Often you use the role itself: engineer, director \ldots.
  On a return envelope you write \jps{行} after your own name; the office
  crosses it out and writes \jps{様}
\end{frame}

\begin{frame}{Honorifics Are Taught}
  \Large
  \begin{itemize}
  \item A mark of \txx{education} and upbringing
  \item Many Japanese want to improve theirs: guides and training abound
  \end{itemize}
  \vfill
  \normalsize
  In my three-week induction at NTT: about 8 hours on honorific language,
  and another 8 or so on who sits where (in a room, a taxi, a train),
  how to pass business cards, and more
\end{frame}

\section{Cultural Scripts}

\begin{frame}{Cultural Scripts}
  \Large
  \begin{itemize}
  \item Norms of communication, written in \txx{semantic primes}:
    \textsc{people}, \textsc{good}, \textsc{bad}, \textsc{say}, \textsc{think}
  \item No culture-specific words, so they compare across cultures
  \end{itemize}
  \vfill
  \normalsize
  Developed by Anna Wierzbicka and colleagues
\end{frame}

\begin{frame}{A Japanese Script: Apologize}
  \begin{nsm}
  \begin{itemize}
  \item {}[people think like this:]
  \item if something bad happens to someone because of me,
  \item I have to say something like this to this person:
    \begin{itemize}
    \item ``I feel something bad because of this''
    \end{itemize}
  \end{itemize}
  \end{nsm}
  \vfill
  \large
  Frequent apologies: they show empathy without accepting blame
\end{frame}

\begin{frame}{Malay: Know the Proper Thing to Do}
  \normalsize
  \begin{tabular}{ll}
    \toprule
    \txx{balasan}     & every deed, good or bad, is repaid in kind \\
    think first       & when I want to do something \\
                      & \quad it is good if I think about it for some time \\
    mind your mouth   & think before speaking, so no one is hurt or shamed \\
    \txx{sensitivity} & avoid blunt or critical speech \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Shaped by Islam and \msa{adat} (custom): ``It is good if a person knows
  what is good to do at all times''.  Western directness can clash with this
\end{frame}

\begin{frame}{Malay: Don't Say ``I Want''}
  \Large
  \begin{tabular}{ll}
    \toprule
    instead of & ``Can you make dinner?'' \\
    say        & ``Are you hungry?'' \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Saying ``I want you to do this'' for your own benefit is discouraged.  The
  listener infers the request, and their autonomy is respected
\end{frame}

\begin{frame}{Malay \msa{menghormati} ``Show Respect''}
  \begin{nsm}
  \begin{itemize}
  \item {}[X \msa{menghormati} Y:]
  \item X thinks good things about Y
  \item X thinks things like this about Y:
    \begin{itemize}
    \item Y is someone above me
    \item I don't want Y to think anything bad about me
    \end{itemize}
  \item X wants Y to know this
  \item because of this, when X is with Y
    \begin{itemize}
    \item X does some things, X doesn't do some other things
    \item X says some things, X doesn't say some other things
    \end{itemize}
  \end{itemize}
  \end{nsm}
\end{frame}

\begin{frame}{Malay: Humility}
  \Large
  \begin{tabular}{ll}
    \toprule
    good & \msa{merendah diri} ``lowering oneself'', humility \\
    bad  & \msa{sombong} ``arrogant'' \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Westerners often find Malay discourse charming and respectful.  Like
  Japanese, it values harmony, patience and deference --- but its roots are
  in Islam.  Modernization is changing these norms
\end{frame}

\section{Indirectness in Japanese}

\begin{frame}{Japanese: Say It Indirectly}
  \large
  \begin{itemize}
  \item Ambiguity avoids confrontation; posture, tone and body language
    carry meaning
  \item Disagreement happens privately, later
  \item \jps{根回し} \jpn[preparing the roots]{nemawashi}: reach consensus
    one by one, before proposing anything
  \end{itemize}
  \vfill
  \Large
  \jpn{Kore-o yaru to ii desu ne.}
  \\ \quad ``It would be good if this were done.''
  \\ \normalsize \quad = my boss telling me ``Do this''
\end{frame}

\begin{frame}{Refusing Without Saying No}
  \Large
  \begin{tabular}{ll}
    \toprule
    \jpn{kentō-shimasu}               & ``I will consider it'' \\
                                      & = probably not \\[1ex]
    \jpn{maemuki-ni kentō-shimasu}    & ``I will consider it positively'' \\
                                      & = I will do nothing \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Refusals come as hesitation and vague language, keeping the peace
\end{frame}

\begin{frame}{Silence and Back-Channelling}
  \large
  \begin{tabular}{ll}
    \toprule
    \multicolumn{2}{l}{\jpn{aizuchi}: show you are listening} \\
    \midrule
    agreement & \jpn{hai} ``yes'', \jpn{sō desu ne} ``so it is'', \jpn{sugoi} ``wow'' \\
    surprise  & \jpn{eeee?} ``really?'', \jpn{hontō desu ka?} ``is that true?'' \\
    grunts    & \jpn{un}, \jpn{nnn} (neutral); \jpn{oo}, \jpn{hoh} (news); \jpn{aa}, \jpn{sa}, \jpn{ma} (stalling) \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Silence is polite: it gives others time to think, and interrupting is rude.
  But a listener should make some sounds, especially on the phone.  I notice
  Siri says \eng{aha} when you speak, so you know it heard you
\end{frame}

\begin{frame}{Deflect the Compliment}
  \Large
  \begin{itemize}
  \item Humility: compliments are \txx{deflected}, not accepted
  \item Too much praise is embarrassing
  \end{itemize}
\end{frame}

\begin{frame}{Culturally Appropriate Communication}
  \Large
  \begin{itemize}
  \item Speaking well means the \txx{appropriate} words, not just the
    right meaning
  \item We rarely notice how our own culture communicates
  \item The more fluent you are, the more people expect the cultural arts too
  \end{itemize}
  \vfill
  \normalsize
  How is Czech communicative culture different from its neighbours'? From
  Asian cultures?
\end{frame}
