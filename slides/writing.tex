\section{Types of Writing Systems}

\begin{frame}{Six Kinds of Script, All in This Region}
  \normalsize
  \begin{tabular}{lll}
    \toprule
    type & a symbol writes \ldots & example \\
    \midrule
    \txx{logographic} & a morpheme                        & Chinese \zhs{山} \cmn{shān} \\
    \txx{abjad}       & a consonant; vowels optional      & Jawi (Arabic script) \\
    \txx{abugida}     & a consonant with an \alert{inherent vowel}; & Thai, Khmer, Burmese \\
                      & \quad other vowels as signs around it & \\
    \txx{alphabet}    & consonants and vowels alike       & Rumi, Vietnamese \\
    \txx{syllabary}   & a syllable (strictly, a mora)     & Japanese kana \\
    \txx{featural}    & features, inside the letter shapes & Korean Hangul \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Japanese \txx{mixes} systems: kanji, two kana, and Latin letters
\end{frame}

\begin{frame}{Three Great Source Traditions}
  \normalsize
  \begin{tabular}{lll}
    \toprule
    Chinese & Brahmi & Arabic \\
    \small c.\ 1200 BCE & \small 3rd c.\ BCE & \small 1200s CE, with Islam \\
    \midrule
    Japanese kanji, kana & Khmer, Thai, Lao   & Malay Jawi \\
    Korean hanja         & Mon, Burmese       & \\
    Vietnamese chữ Nôm   & Javanese, Balinese & \\
    \bottomrule
  \end{tabular}

  \vfill
  \normalsize
  Brahmi reached Southeast Asia through the south Indian Pallava script.
  Then two newcomers: the \txx{Latin} alphabet with missionaries and
  colonial rule (Rumi, Vietnamese), and an \txx{invention}: Korean Hangul
\end{frame}

\begin{frame}{Where Did Brahmi Come From?}
  \large
  \begin{itemize}
  \item First securely dated under \txx{Ashoka}, 3rd century BCE
  \item Its neighbour \txx{Kharoṣṭhī} (right to left) clearly comes from
    \txx{Aramaic}, the Persian empire's administrative script
  \item Brahmi most likely had a Semitic model too, \txx{reworked} by Indian
    scholars: left to right, vowel signs, long vs short vowels
  \item Descent from the Indus script is a minority view: that script is
    undeciphered and 1,500 years older
  \end{itemize}
  \vfill
  \Large
  \begin{tabular}{lllll}
    Devanagari & Thai & Lao & Khmer & Burmese \\
    \mtplain{\devafont क} & \mtplain{\textthai{ก}} & \mtplain{\laofont ກ} &
    \mtplain{\khmerfont ក} & \mtplain{\myanmarfont က} \\
  \end{tabular}
  \quad\normalsize all \emph{ka}
  \vfill
  \small \citet{salomon:1998,scharfe:2002}
\end{frame}

\section{Abjad: Jawi}

\begin{frame}{Jawi: Arabic Script for Malay}
  \large
  \begin{tabular}{ll}
    \textarabic{السلام عليكم} & \ar{as-salām 'alaykum} ``peace be upon you'' \\
    \textarabic{جاوي}         & \msa{Jawi} $<$ \ar{al-Jazā'ir al-Jāwī} ``Java archipelago'' \\
  \end{tabular}
  \vfill
  \normalsize
  \begin{itemize}
  \item An \txx{abjad}: written right to left, short vowels usually not written
  \item \msa{Jawi} also means ``Malay'': \msa[Malay language]{Bahasa Jawi}
  \end{itemize}
\end{frame}

\section{Abugida: Thai}

\begin{frame}{Thai: An Abugida}
  \Large
  \begin{tabular}{lll}
    \textthai{คน}    & \tha{khon}     & person: \alert{no vowel written} \\
    \textthai{สวัสดี} & \tha{sawàtdii} & hello \\
  \end{tabular}
  \vfill
  \normalsize
  \begin{itemize}
  \item Each consonant carries an \txx{inherent vowel}; other vowels are signs
    above, below, before or after it
  \item No spaces between words
  \item From Old Khmer, which goes back to Brahmi via Pallava
  \end{itemize}
\end{frame}

\begin{frame}{Thai Tones: Letter Class Matters}
  \Large
  \begin{tabular}{llllll}
    \toprule
    \multicolumn{3}{l}{high-class \textthai{ข}} & \multicolumn{3}{l}{low-class \textthai{ค}} \\
    \midrule
    \textthai{ขา}  & \tha{khǎa}  & leg      & \textthai{คา}  & \tha{khaa}  & embedded \\
    \textthai{ข่า} & \tha{khàa}  & galangal & \textthai{ค่า} & \tha{khâa}  & cost \\
    \textthai{ข้า} & \tha{khâa}  & slave    & \textthai{ค้า} & \tha{kháa}  & trade \\
    \textthai{ขาด} & \tha{khàat} & lack     & \textthai{คาด} & \tha{khâat} & strap \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Same /kh/, same tone mark, different tone: the tone depends on the
  consonant's class, the tone mark and the syllable.  Complicated but
  deterministic (Smalley 1994: 184)
\end{frame}

\begin{frame}{Hangul: Invented, Not Inherited}
  \Large
  \textkorean{안녕하세요} \ko[hello]{annyeonghaseyo}
  \vfill
  \large
  \begin{itemize}
  \item Invented by \txx{King Sejong} in the 15th century, and documented in
    the \textit{Hunmin jeongeum} ``The correct sounds for the instruction of
    the people''
  \item Looked down on by scholars, who used \txx{hanja}, as a script for
    women, children and the uneducated
  \item Not widely adopted until the 19th century
  \end{itemize}

\section{Logographic: Chinese}

\begin{frame}{Chinese: 3,000 Years and Counting}
  \Large
  \begin{tabular}{lll}
    \zhs{山}   & \cmn{shān}  & mountain \\
    \zhs{你好} & \cmn{nǐ hǎo} & hello (``you good'') \\

  \vfill
  \large
  \begin{itemize}
  \item Grew out of pictograms; in continuous use for over 3,000 years
  \item Characters mostly write \txx{morphemes}

  \item Simplified in the People's Republic in the 1950s
  \end{itemize}

\begin{frame}{Chinese Characters for Other Sinitic Languages}
  \large
  \begin{itemize}
  \item The script began with \txx{Old Chinese}, c.\ 1200 BCE
  \item The modern standard written language is based on \txx{Mandarin}, so
    phonetic hints fit Mandarin, not Cantonese or Hokkien
  \item Literacy in Cantonese or Taiwanese assumes the Mandarin standard
  \item Many Cantonese words and particles have no standard character
  \end{itemize}
\end{frame}

  \vfill
  \normalsize
  Kanjidic gives \jps{明} 3 on, 10 kun and 11 name readings; Chinese and
  Korean have one each: \cmn{míng}, \kor{myeong}.
  \url{https://www.edrdg.org/cgi-bin/wwwjdic/wwwjdic?1B}
\end{frame}

\begin{frame}{Why So Many Readings?}
  \Large
  \begin{itemize}
  \item Kanji arrived from the 5th century, \txx{with} Chinese pronunciations:
    \txx{on} readings, in waves from different periods and regions
  \item Many already had a native word: \txx{kun} readings
  \item Compounds tend to use on (\jps{火山} \jpn{ka-zan} ``volcano''),
    single kanji kun (\jps{山} \jpn{yama})
  \end{itemize}
\end{frame}

\begin{frame}{The Cost: Memory, and Reforms}
  \Large
  \begin{tabular}{ll}
    \toprule
    1946 & \txx{Tōyō} kanji: 1,850 \\
    1981 & \txx{Jōyō} kanji: 1,945 \\
    2010 & \txx{Jōyō} kanji, revised: 2,136 \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  \begin{itemize}
  \item An educated adult knows about 2,000 kanji, each with several
    readings: years of study
  \item Names and literature still use rarer kanji
  \item Typing means choosing among homophones; handwriting skills suffer
  \end{itemize}
\end{frame}

\section{Unicode}

\begin{frame}{Unicode: Code Points Are Not Glyphs}
  \large
  \begin{itemize}
  \item Unicode gives each character a \txx{code point}; the \txx{font} draws it
  \item \txx{Han unification}: Chinese, Japanese and Korean share code points,
    though local shapes differ
  \end{itemize}
  \vfill
  \Huge
  \begin{tabular}{ccccc}
    \zhs{骨} & \zht{骨} & \jps{骨} & \kos{骨} \\
    \normalsize Simplified & \normalsize Traditional & \normalsize Japanese & \normalsize Korean \\
  \end{tabular}
  \quad \normalsize all U+9AA8
  \vfill
  \large
  \begin{itemize}
  \item Rare characters need a font that has them: obsolete \kos{ㅸ}
    for Cia-Cia, and chữ Nôm (Extension B), which our fonts lack: \tofu
  \end{itemize}
\end{frame}

\begin{frame}{Summary}
  \Large
  \begin{itemize}
  \item The region has \txx{every type} of script: logographic, abjad,
    abugida, alphabet, syllabary, featural
  \item Most descend from \txx{three traditions}: Chinese, Brahmi, Arabic ---
    plus Latin, and one invention
  \item Scripts are \txx{hard to change}: most countries have dropped or
    limited Chinese characters, but reprinting everything is a massive task
  \end{itemize}
