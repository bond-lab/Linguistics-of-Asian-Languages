\section{Loans as a map of the past}

\begin{frame}{Loanwords Are a Map of Contact}
  \Large
  \begin{itemize}
  \item Every borrowed word records a \txx{contact}: trade, religion,
    conquest, fashion
  \item Old loans blend in; new ones still look \txx{foreign}
  \item Peel back the layers to read a language's \txx{history}
  \end{itemize}
  \vfill
  \normalsize
  This week follows \citet[Chapter 3]{goddard:2005}.
\end{frame}

\begin{frame}{Four Kinds of Loan}
  \large
  \begin{tabular}{lll}
    \toprule
    Kind & Example & Comment \\
    \midrule
    \txx{indigenized} & \eng{gift}, \eng{root} & so old nobody notices \\
    \txx{prestige}    & \eng{crown}, \eng{judge} & high status, like the rulers' language \\
    \txx{cultural}    & \eng{coffee}, \eng{sushi} & a new thing arrives \\
    \txx{calque}      & \eng{sky-scraper} & translate the parts: \\
                      & & German \mtciteform{Wolken-kratzer} \\
                      & & French \fra{gratte-ciel} \\
    \bottomrule
  \end{tabular}
\end{frame}

\section{Building words --- Derivational Morphology}

\begin{frame}{Three Ways to Build a Word}
  \Large
  \begin{tabular}{ll}
    \toprule
    \txx{affixation}   & \eng{un-reli-able}, \eng{sad-ness}, \eng{re-written} \\
    \txx{compounding}  & \eng{book-shop}, \eng{black-board} \\
    \txx{reduplication} & \eng{choo-choo} \\
    \bottomrule
  \end{tabular}

  \vfill
  \normalsize
  \begin{itemize}
  \item \txx{Derivation} makes new words; \txx{inflection} changes a word's
    form for grammar (tense, number)
  \item Sinitic and mainland Southeast Asian languages have almost no
    inflection; Japanese, Korean and Austronesian do
  \item Derivation is more common than inflection in many Asian languages
  \end{itemize}
\end{frame}

\begin{frame}{Compounding: The Main Way to Make Words}
  \Large
  \begin{itemize}
  \item Sinitic and mainland Southeast Asian languages rely on it
  \item Usually \txx{two} elements: nouns, adjectives or verbs
  \item Productive, but the meaning is not always predictable
  \end{itemize}
\end{frame}

\section{Meaning differences between languages}

\begin{frame}{Some Words Have No Equivalent}
  \Large
  \begin{tabular}{lll}
    \jps{お見合い} & \jpn{omiai}    & formal marriage meeting \\[1ex]
    \jps{酒}       & \jpn{sake}     & \alert{any} alcoholic drink \\
    \jps{日本酒}   & \jpn{nihonshu} & ``Japanese sake'' --- the rice wine \\
  \end{tabular}

  \vfill
  \normalsize
  \begin{itemize}
  \item English borrowed \eng{sake} for the rice wine,
    but in Japanese it has broadened to all alcohol
  \item Words for emotions, values and ideals differ too --- and so do
    basic words
  \end{itemize}
\end{frame}

\begin{frame}{Describing Meaning: Polysemy}
  \Large
  How can we describe it when we don't have it in our language?
  
  \begin{itemize}
  \item Define with \txx{simpler} words, never circular or obscure ones
  \item \txx{Polysemy}: one word, several related meanings
  \item Languages \txx{group} meanings differently
    --- so compare the meanings, not just the words
  \end{itemize}
\end{frame}

\begin{frame}{Languages Carve Up Meanings Differently}
  \large
  \begin{tabular}{llllll}
    \toprule
    English & Czech & Malay & Japanese & Mandarin & \\
    \midrule
    \eng{nose}   & \mtciteform{nos}    & \msa{hidung}  & \multirow{3}{*}{\jpn{hana}} & \multirow{2}{*}{\cmn{bízi}} & of a person \\
    \eng{trunk}  & \mtciteform{chobot} & \msa{belalai} &                             &                             & of an elephant \\
    \eng{muzzle} & \mtciteform{čenich} & \msa{muncung} &                             & \multirow{2}{*}{\cmn{zuǐ}}  & of a dog \\
    \eng{beak}   & \mtciteform{zobák}  & \msa{paruh}   & \jpn{kuchibashi}            &                             & of a bird \\
    \bottomrule
  \end{tabular}

  \bigskip
  \begin{tabular}{lll}
    \toprule
    English & Malay & \\
    \midrule
    \multirow{2}{*}{\eng{live}} & \msa{hidup}   & be alive \\
                                & \msa{tinggal} & reside somewhere \\
    \bottomrule
  \end{tabular}
\end{frame}

\begin{frame}{The Syntax Can be Different}
  \Large
  \begin{itemize}
  \item Each meaning comes with its own \txx{grammatical frame}
  \item English \eng{live} ``reside'' needs a place: \eng{live \alert{in Prague}}
  \item Malay \msa{buat}: \msa[make]{buat kuih} vs \msa[do]{buat kerja}
  \end{itemize}
  \vfill
  Different arguments $\Rightarrow$ different meanings
\end{frame}

\begin{frame}{``Exist (There is)'' and ``Have'': Often One Word}
  % gb4e resets the size, so set it through its own hooks
  \renewcommand{\exfont}{\Large}
  \renewcommand{\exnrfont}{\Large}
  \renewcommand{\glossfont}{\Large}
  \renewcommand{\transfont}{\Large}
  \renewcommand{\eachwordtwo}{\upshape}
  \glossglue = 6pt plus 2pt minus 1pt
  \begin{exe}
    \ex
    \gll Ada dua ekor lembu. \\
         exist two \textsc{clf} cow \\
    \glt ``There are two cows.'' [zsm]
    \ex
    \gll \ul{Orang} \ul{ini} ada dua ekor lembu. \\
         person this exist two \textsc{clf} cow \\
    \glt ``This person has two cows.'' [zsm]
  \end{exe}

  \vfill
  \normalsize
  Malay \msa{ada}, Mandarin \cmn{yǒu}: one argument means ``there is'',
  two arguments mean ``have''
\end{frame}

\begin{frame}{Summary: Meaning Differs, Even for Basic Words}
  \Large
  \begin{itemize}
  \item Some words have \txx{no equivalent}
  \item Even basic words \txx{split or merge} meanings differently
  \item Kinship, rice and emotion words show what a culture \txx{cares about}
  \end{itemize}
\end{frame}

\section{Some cultural key words}

\begin{frame}{Cultural Key Words}
  \Large
  \begin{itemize}
  \item Words around which a culture \txx{organizes} a whole domain
  \item Common, and turn up in proverbs and sayings
  \item Often have \txx{no equivalent} elsewhere
  \end{itemize}
  \vfill
  \begin{tabular}{ll}
    English    & \eng{love}, \eng{honesty} \\
    Australian & \eng{mate} \\
    Japanese   & \jpn{omiai} \\
  \end{tabular}
  \vfill
\end{frame}

\begin{frame}{Describing Them Without Bias}
  \large
  \begin{itemize}
  \item English glosses smuggle in English concepts (\txx{ethnocentrism})
  \item Instead, explain with \txx{semantic primes}: simple meanings found
    in every language \citep{wierzbicka:1996}
  \end{itemize}
  \vfill
  \begin{center}
    \LARGE
    \textsc{someone} \quad \textsc{think} \quad \textsc{good} \quad
    \textsc{want} \quad \textsc{because}
  \end{center}
  \vfill
  \large
  This \txx{Natural Semantic Metalanguage} (NSM) can be checked
  against native speakers' intuitions
\end{frame}

\begin{frame}{Malay \msa{malu}}
  \Large
  ``shame'', ``shy'', ``embarrassed''
  \vfill
  \begin{tabular}{ll}
    \msa{Aku malu.}            & ``I'm ashamed.'' \\
    \msa{Eh! Malu pulak dia!}  & ``Oh, now they're shy!'' \\
  \end{tabular}
  \vfill
  \normalsize
  Fear that others will think or say something bad about you
  $\Rightarrow$ wanting to avoid them; it keeps people in line
\end{frame}

\begin{frame}{Explication: \msa{malu}}
  \begin{nsm}
  \begin{itemize}
    \item X feels \msa{malu}
    \item X feels something bad because X thinks like this:
      \begin{itemize}
      \item people can know something about me
      \item people can think something bad about me because of this
      \item people can say something bad about me because of this
      \item I don't want this
      \end{itemize}
    \item because X feels like this, X doesn't want to be near other people
    \item it is good if people can feel something like this
  \end{itemize}
  \end{nsm}
\end{frame}

\begin{frame}{Malay \msa{sabar}}
  \Large
  ``patience'' --- but broader: staying calm, enduring without losing control
  \vfill
  \large
  \begin{tabular}{ll}
    \msa{Bersabar\ldots}            & ``Be patient\ldots'' \\
    \msa{Sabar, jangan menangis.}   & ``Calm down, don't cry.'' \\
  \end{tabular}
  \vfill
  \msa{Allah sentiasa bersama dengan orang-orang yang sabar}
  \\ ``Allah is always with those who are patient''
\end{frame}

\begin{frame}{Explication: \msa{sabar}}
  \begin{nsm}
  \begin{itemize}
  \item X is \msa{sabar} [at the time]
  \item at this time, X felt something bad
  \item because of this, X could have thought:
    \begin{itemize}
    \item I don't want this
    \item I want to do something now
    \end{itemize}
  \item X did not think like this, because X didn't want to think like this
  \item it is good if a person can be like this
  \end{itemize}
  \end{nsm}
  \vfill
  \msa{Sabar} takes inner discipline and self-control
\end{frame}

\begin{frame}{Chinese \zhs{孝} \cmn{xiào}}
  \Large
  filial piety: duty, respect and care for parents, alive or dead
  \vfill
  \large
  \begin{tabular}{l}
    \zhs{百善孝为先} \cmn{bǎi shàn xiào wéi xiān} \\
    ``Of the hundred good deeds, \cmn{xiào} comes first'' \\[2ex]
    \zhs{父母恩比山高，比海深} \cmn{fùmǔ ēn bǐ shān gāo, bǐ hǎi shēn} \\
    ``What our parents give us is higher than a mountain, \\
    deeper than the ocean'' \\
  \end{tabular}
  \vfill
\end{frame}

\begin{frame}{Explication: \zhs{孝} \cmn{xiào} (1)}
  \begin{nsm}
  \begin{itemize}
  \item everyone can think about some other people like this:
    \begin{itemize}
    \item this person is my father, this person is my mother
    \item I exist because of them
    \item because of this when I think about them I feel something very good
    \item I cannot think about other people like this
    \end{itemize}
  \item it is good if a person thinks about these people at all times
    \begin{itemize}
    \item it is good if this person feels something because of this
    \end{itemize}
  \end{itemize}
  \end{nsm}
\end{frame}

\begin{frame}{Explication: \zhs{孝} \cmn{xiào} (2)}
  \begin{nsm}
  \begin{itemize}
  \item it is good if a person thinks about these people like this:
    \begin{itemize}
    \item I want them to feel something very good at all times
    \item because of this I have to do many good things for them
    \item I want to do these things
    \item I don't want these people to feel anything bad at any time
    \item because of this I cannot do some things
    \item I don't want to do these things
    \end{itemize}
  \item it is good if a person thinks like this about some people
    \begin{itemize}
    \item it is good if a person does many things because of this
    \item it is bad if a person doesn't think like this about some people
    \end{itemize}
  \end{itemize}
  \end{nsm}
\end{frame}

\begin{frame}{Chinese \zhs{忍} \cmn{rěn}}
  \Large
  perseverance: enduring hardship without complaint, for a greater goal
  \vfill
  \zhs{忍气吞声} \cmn{rěn qì tūn shēng}
  \\ ``endure one's anger and swallow one's voice''
  \vfill
  \normalsize
  Long-term goals over immediate desires; a source of inner strength
\end{frame}

\begin{frame}{Explication: \zhs{忍} \cmn{rěn} (1)}
  \begin{nsm}
  \begin{itemize}
  \item everyone can think like this about some things:
    \begin{itemize}
    \item I want these things to happen
    \item I know they cannot happen if I don't do some things
    \item I want to do these things because of this
    \item if I have to do these things for a long time I don't want not to
      do them because of this
    \item if I feel something bad when I do these things I don't want not
      to do them because of this
    \end{itemize}
  \end{itemize}
  \end{nsm}
\end{frame}

\begin{frame}{Explication: \zhs{忍} \cmn{rěn} (2)}
  \begin{nsm}
  \begin{itemize}
  \item it is good if a person always thinks like this
  \item it is good if a person thinks about this when this person feels
    something bad
  \item it is good if a person thinks about this when someone else does
    something bad to this person
  \item it is good if a person does many things because this person thinks
    like this
  \item it is good if a person can be like this
  \end{itemize}
  \end{nsm}
\end{frame}

\begin{frame}{Japanese \jps{甘え} \jpn{amae}}
  \Large
  dependence: taking someone's love and care for granted
  \vfill
  ``When I am with Y, nothing bad can happen to me''
  \vfill
  \normalsize
  Modelled on an infant's dependence on its mother; part of all close
  relationships in Japan
\end{frame}

\begin{frame}{Explication: \jps{甘え} \jpn{amae}}
  \begin{nsm}
  \begin{itemize}
  \item X feels \jpn{amae} towards Y
  \item X thinks like this about Y:
    \begin{itemize}
    \item when Y thinks about me, Y feels something good
    \item Y wants to do good things for me
    \item Y can do good things for me
    \item when I am with Y nothing bad can happen to me
    \item I don't have to do anything because of this
    \item I want to be with Y
    \end{itemize}
  \item X feels something good because of this
  \end{itemize}
  \end{nsm}
\end{frame}

\begin{frame}{Japanese \jps{思い遣り} \jpn{omoiyari}}
  \Large
  empathy: knowing what others need without being told
  \\ \normalsize literally ``think-give''
  \vfill
  \large
  \begin{tabular}{l}
    \jpn{Omoiyari no aru ko ni natte ne.} \\
    ``Please become a person with \jpn{omoiyari}.'' \\[2ex]
    \jpn{Omoiyari no kokoro o taisetsu ni shimashō.} \\
    ``Let's treasure the heart of \jpn{omoiyari}.'' \\
  \end{tabular}
  \vfill
  \normalsize
  Keeps harmony and avoids conflict
\end{frame}

\begin{frame}{Explication: \jps{思い遣り} \jpn{omoiyari}}
  \begin{nsm}
  \begin{itemize}
  \item X has \jpn{omoiyari}
  \item X often thinks like this about other people:
    \begin{itemize}
    \item I can know what this person feels
    \item I can know what this person wants
    \item I can do something good for this person because of this
    \item I want to do this
    \item this person doesn't have to say anything
    \end{itemize}
  \item because of this, X does something
  \item people think this is good
  \end{itemize}
  \end{nsm}
\end{frame}

\begin{frame}{Conclusions}
  \Large
  \begin{itemize}
  \item \txx{Loans} map contact history: English, Malay and Japanese are
    built in layers
  \item Words are built by \txx{compounding}, \txx{reduplication} and
    \txx{affixation}, in different mixes
  \item Languages \txx{carve up meaning} differently, even for water,
    rice and siblings
  \item \txx{Cultural key words} can be explained without bias in
    semantic primes
  \end{itemize}
\end{frame}
