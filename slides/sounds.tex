\section{The Big Picture}

\begin{frame}{Five Sound Profiles}
  \large
  \begin{tabular}{llll}
    \toprule
    & inventory & words & pitch \\
    \midrule
    insular SE Asia  & small            & disyllables & --- \\
    mainland SE Asia & large, aspiration & monosyllables & \txx{tone} \\
    Sinitic          & initials + finals          & monosyllables     & \txx{tone} \\
    Korean           & three-way stops            & complex codas     & --- \\
    Japanese         & small                      & morae             & \txx{pitch accent} \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Phoneme counts are analyses, not facts: analysts disagree on what is a
  phoneme, the same sound gets different symbols, and spelling often
  differs from IPA
\end{frame}

\section{Insular Southeast Asia}

\begin{frame}{Malay Consonants}
  \large
  \begin{tabular}{llllll}
    \toprule
    & labial & alveolar & palatal & velar & glottal \\
    \midrule
    stops       & p b & t d & c j        & k g        & [ʔ] \\
    nasals      & m   & n   & ny [ɲ]     & ng [ŋ]     & \\
    fricatives  &     & s   &            &            & h \\
    lateral     &     & l   &            &            & \\
    trill       &     & r   &            &            & \\
    semivowels  & w   &     & y [j]      &            & \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  18 native consonants; [ʔ] began as an allophone of /k/.  Loans add f, v,
  z, sy [ʃ], kh [x] in educated speech; ny, ng, sy, kh are single phonemes
\end{frame}

\begin{frame}{Malay: How a Glottal Stop Becomes a Phoneme}
  \Large
  \begin{tabular}{lll}
    \toprule
    \msa{kakak}  & [kakaʔ] & older sister \\
    \msa{rakyat} & [raʔjat] & citizenry \\
    \midrule
    \msa{bisik}  & [bisiʔ] & whisper \\
    \msa{fisik}  & [fisik] & physical (a loan) \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Word-final /k/ is [ʔ] in native words, but loans keep [k]: a new contrast.
  Spelling writes ⟨k⟩ for both, hiding it.  Allophone $\to$ phoneme,
  through borrowing
\end{frame}

\begin{frame}{Vowels: Tagalog, Malay, Acehnese}
  \Large
  \begin{tabular}{lll}
    \toprule
    Tagalog & Malay & Acehnese \\
    \midrule
    i \quad u & i \quad u       & i \quad ɯ \quad u \\
    e \quad o & e \quad ə \quad o & e \quad ə \quad o \\
    a         & a               & ɛ \quad ʌ \quad ɔ \\
              &                 & a \\
    \midrule
    5 & 6 & 10 \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Malay adds a central schwa (also written ɨ, ɜ).  Acehnese, on Sumatra, has
  ten vowels plus diphthongs: more like Mon-Khmer than Austronesian, a sign
  of contact
\end{frame}

\begin{frame}{Insular Word Shapes: Two Syllables}
  \Large
  \begin{tabular}{ll}
    \toprule
    template      & (C)V\textsubscript{1}(C)(C)V\textsubscript{2}(C) \\
    monosyllables & mostly exclamations and clippings \\
    longer words  & compounds, derivations, loans \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  \begin{itemize}
  \item Eastern Indonesia prefers simpler CVCV roots
  \item Malay final syllables: high vowels when open, mid (e, o) when closed
  \item Some dialects lose final consonants: only [ʔ] remains, nasals
    merge to [ŋ]
  \end{itemize}
\end{frame}

\section{Mainland Southeast Asia}

\begin{frame}{Mainland: Big Inventories, Aspiration, Tone}
  \Large
  \begin{itemize}
  \item 30--40+ phonemes
  \item \txx{Aspiration} contrasts; voicing often does not
  \item Large vowel systems, especially Mon-Khmer
  \item Tone, often born from lost consonant contrasts (\txx{tonogenesis})
  \end{itemize}
\end{frame}

\begin{frame}{Thai Consonants}
  \large
  \begin{tabular}{llllll}
    \toprule
    & labial & alveolar & palatal & velar & glottal \\
    \midrule
    plain stops     & p   & t   & c   & k   & ʔ \\
    aspirated stops & pʰ  & tʰ  & cʰ  & kʰ  & \\
    voiced stops    & b   & d   &     &     & \\
    nasals          & m   & n   &     & ŋ   & \\
    fricatives      & f   & s   &     &     & h \\
    liquids         &     & r, l &    &     & \\
    semivowels      & w   &     & j   &     & \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Three stop series: plain, aspirated, voiced.  /b, d/ are often
  preglottalized [ɓ, ɗ].  Romanizations write ph, th, kh (not f, θ, x!)
\end{frame}

\begin{frame}{Thai Vowels}
  \Large
  \begin{tabular}{llll}
    \toprule
    & front & back unrounded & back rounded \\
    \midrule
    high & i & \alert{ɯ} & u \\
    mid  & e & \alert{ɤ} & o \\
    low  & ɛ & a         & ɔ \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Nine vowels, each \txx{short or long} (/khaw/ vs /khaaw/), plus diphthongs.
  Back \emph{unrounded} vowels are unusual in Europe but common here
  (Hudak 1987, in Goddard 2005)
\end{frame}

\begin{frame}{Burmese, Lahu, Khmer, Hmong}
  \large
  \begin{tabular}{ll}
    \toprule
    Burmese & about 34 consonants: plain, aspirated, voiced stops; \\
            & \quad \alert{voiceless} nasals and approximants: hm, hn, hl, hw \\
    Lahu    & no voiceless nasals; post-velar stops; and no /s/! \\
    Khmer   & 27 vocalic nuclei (Battambang): length, diphthongs \\
    Hmong   & four stop series (prenasalized $\times$ aspirated), \\
            & \quad six places of articulation: among the world's largest \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Some Mon-Khmer languages have 40 or more vocalic nuclei
\end{frame}

\begin{frame}{Vietnamese: Six Tones}
  \large
  \begin{tabular}{lllll}
    \toprule
    tone & & contour & example & \\
    \midrule
    \vie{ngang} & level        & 33 \tl{˧}     & \vie{ma}  & ghost \\
    \vie{huyền} & low falling  & 21 \tl{˨˩}    & \vie{mà}  & but \\
    \vie{sắc}   & rising       & 35 \tl{˧˥}    & \vie{má}  & cheek, mother \\
    \vie{hỏi}   & dipping      & 313 \tl{˧˩˧}  & \vie{mả}  & tomb \\
    \vie{ngã}   & broken, creaky & 3ʔ5    & \vie{mã}  & horse \\
    \vie{nặng}  & low, glottal & 21ʔ      & \vie{mạ}  & rice seedling \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Northern Vietnamese.  The diacritics are the ones you met in week 8.
  About 23 consonants, voiced fricatives and retroflexes; syllables
  (C)(w)V(V)(C)
\end{frame}

\begin{frame}{Voice Quality: Phonation}
  \large
  \begin{tabular}{lll}
    \toprule
    \txx{modal}   & ordinary voicing             & \\
    \txx{breathy} & extra airflow      & Mon, older Khmer register \\
    \txx{creaky}  & tense, irregular   & Vietnamese \vie{ngã}, \vie{nặng}; \\
                  &                              & \quad Burmese creaky tone \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  In \txx{register} languages (Mon, older Khmer), a clear vs breathy voice
  contrast does the work tone does elsewhere.  Phonation, pitch and vowel
  quality are tied together --- and are where tone comes from
\end{frame}

\section{Tone}

\begin{frame}{Writing Tones: Chao Numbers}
  \Large
  \begin{tabular}{lllll}
    \toprule
    \multicolumn{5}{l}{pitch from 1 (lowest) to 5 (highest)} \\
    \midrule
    \zhs{妈} & \cmn{mā} & high level   & 55  & \tl{˥} \\
    \zhs{麻} & \cmn{má} & rising       & 35  & \tl{˧˥} \\
    \zhs{马} & \cmn{mǎ} & dipping      & 214 & \tl{˨˩˦} \\
    \zhs{骂} & \cmn{mà} & falling      & 51  & \tl{˥˩} \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Mandarin: mother, hemp, horse, scold.  The numbers trace the pitch through
  the syllable; the tone letters draw it.  A third of the world's languages
  have tone
\end{frame}

\begin{frame}{Thai: Five Tones}
  \Large
  \begin{tabular}{llll}
    \toprule
    \ths{คา}  & \tha{khaa}  & mid     & be lodged in \\
    \ths{ข่า} & \tha{khàa}  & low     & galangal \\
    \ths{ข้า} & \tha{khâa}  & falling & servant \\
    \ths{ค้า} & \tha{kháa}  & high    & trade \\
    \ths{ขา}  & \tha{khǎa}  & rising  & leg \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  The syllable type limits the tones: short vowels and checked syllables
  allow fewer contours.  No neutral tone, unlike Mandarin
\end{frame}

\begin{frame}{Where Tone Comes From}
  \Large
  \begin{itemize}
  \item Pitch is how we hear vocal-fold frequency (F0)
  \item Voicing lowers pitch, voicelessness raises it
  \item Lose the voicing contrast, keep the pitch: a new \txx{tone}
  \end{itemize}
  \vfill
  \normalsize
  Stiff and creaky voice often go with low tones (as in Cantonese).
  This is \txx{tonogenesis}
\end{frame}

\begin{frame}{Allotones vs.\ Sandhi}
  \large
  \begin{tabular}{ll}
    \toprule
    \multicolumn{2}{l}{\txx{allotone}: a tone's variant, chosen by its environment} \\
    \midrule
    Mandarin    & 214 $\to$ 21 before another tone \\
    Cantonese   & ``9 tones'' = 6 + 3 checked (shortened level tones) \\
    a Lao dialect & 33 and 213 in complementary distribution \\
    \midrule
    \multicolumn{2}{l}{\txx{sandhi}: a tone changes because of a neighbouring tone} \\
    \midrule
    Mandarin    & 214 214 $\to$ 35 214: \zhs{你好} \cmn{nǐ hǎo} $\to$ [ní hǎo] \\
    Tianjin     & 21 21 $\to$ 13 21;\enspace 53 53 $\to$ 21 53 \\
    Cantonese   & almost none \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Mandarin also has a \txx{neutral tone} on weak syllables (\cmn{de})
\end{frame}

\begin{frame}{Asian Tone Is Different}
  \large
  \begin{tabular}{ll}
    \toprule
    East and Southeast Asia & many \txx{contour} tones, one per syllable \\
    Africa, Mesoamerica     & mostly \txx{level} tones, spread over words \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  African tone inspired autosegmental phonology, where contours are
  sequences of levels (214 = L + H)
\end{frame}

\section{Sinitic}

\begin{frame}{Mandarin: Initials and Finals}
  \Large
  \begin{tabular}{ll}
    \toprule
    initial & 21 consonants (pinyin); aspiration, retroflexes \\
    final   & medial + vowel + coda: \cmn{-ao}, \cmn{-iao}, \cmn{-uai}, \cmn{-ui} \\
    coda    & only /n/ and /ŋ/ \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Template (C)(G)V(V/N).  Chinese tradition analyses the syllable as
  initial + final, not as onset + rhyme.  j, q, x are arguably allophones
  of other initials
\end{frame}

\begin{frame}{Cantonese Initials}
  \large
  \begin{tabular}{llllll}
    \toprule
    & labial & alveolar & velar & labio-velar & glottal \\
    \midrule
    plain stops     & p  & t  & k  & kʷ  & \\
    aspirated stops & pʰ & tʰ & kʰ & kʷʰ & \\
    affricates      &    & ts, tsʰ &  &  & \\
    fricatives      & f  & s  &    &     & h \\
    nasals          & m  & n  & ŋ  &     & \\
    approximants    & w  & l, j &  &     & \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  19 initials (with zero), plus syllabic nasals \zht{唔} \yue{m̀h},
  \zht{五} \yue{ńgh}.  Six tones; the ``nine'' count adds three checked
  allotones
\end{frame}

\section{Korean and Japanese}

\begin{frame}{Korean: Three-Way Stops}
  \Large
  \begin{tabular}{llll}
    \toprule
    lax       & \kos{불} & \IPA{pul}  & fire \\
    aspirated & \kos{풀} & \IPA{pʰul} & grass \\
    tense     & \kos{뿔} & \IPA{p͈ul}  & horn \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Lax stops are voiceless (often heard as voiced); tense ones are glottalized,
  with high muscular tension.  /r/ is [l] in codas, [ɾ] between vowels
\end{frame}

\begin{frame}{Korean Vowels}
  \Large
  \begin{tabular}{llll}
    \toprule
    & front & back unrounded & back rounded \\
    \midrule
    high & i & ɯ & u \\
    mid  & e & ʌ & o \\
    low  & ɛ & a & \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  About eight simple vowels; front rounded ø, y are mostly diphthongs now.
  Again a high back unrounded [ɯ]
\end{frame}

\begin{frame}{Japanese: A Small System, Counted in Morae}
  \Large
  \begin{tabular}{ll}
    \toprule
    \jpn{hachimaki} & ha-chi-ma-ki \\
    \jpn{shinbun}   & shi-\alert{N}-bu-\alert{N} \\
    \jpn{yappari}   & ya-\alert{Q}-pa-ri \\
    \bottomrule
  \end{tabular}
  \quad all four morae
  \vfill
  \normalsize
  Five vowels, few consonants.  The \txx{mora} is the timing unit of poetry
  and kana; N (moraic nasal) and Q (first half of a geminate) each count
  one.  High vowels devoice between voiceless consonants: \jpn{desu} [desu̥]
\end{frame}

\section{Syllables}

\begin{frame}{Syllable Shapes Compared}
  \large
  \begin{tabular}{ll}
    \toprule
    insular SE Asia & disyllabic roots; CVCV in the east \\
    Mon-Khmer       & \txx{sesquisyllables}: a minor + a major syllable \\
    Sinitic         & initial + final; only nasal codas \\
    Vietnamese      & (C)(w)V(V)(C); codas nasal or voiceless stop \\
    Korean          & no onset clusters; written coda clusters \\
                    & \quad simplified in speech: \kos{닭} [tak] ``chicken'' \\
    Japanese        & (C)V, with only N and Q as codas \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  So Japanese adapts loans heavily: \eng{strike} $\to$ \jpn{sutoraiku}
\end{frame}

\begin{frame}{Sesquisyllables: One and a Half Syllables}
  \Large
  \begin{tabular}{ll}
    \toprule
    Khmer   & C(r)V\textsubscript{1}(N)\,--\,C(C)V\textsubscript{2}(C) \\
    Burmese & C\textsubscript{1}ə\,--\,C\textsubscript{2}(w/y)V(C\textsubscript{3}) \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  The minor syllable is unstressed, with few vowels; the major one has rich
  clusters and unusual onsets.  Burmese took the pattern from Mon-Khmer
\end{frame}

\section{Morphophonemics}

\begin{frame}{Morphemes Change Shape}
  \Large
  \begin{tabular}{lll}
    \toprule
    \eng{cat-s}  & [s]  & after voiceless \\
    \eng{dog-s}  & [z]  & after voiced \\
    \eng{hors-es} & [əz] & after sibilants \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  An underlying form plus rules: assimilation, deletion, insertion,
  coalescence --- often by copying a feature from a neighbour
\end{frame}

\begin{frame}{Malay \msa{meN-}: The Nasal Agrees}
  \Large
  \begin{tabular}{lll}
    \toprule
    \msa{baca}  & \msa{mem-baca}   & read \\
    \msa{pukul} & \msa{me\alert{m}ukul}   & hit \\
    \msa{tulis} & \msa{me\alert{n}ulis}   & write \\
    \msa{sapu}  & \msa{me\alert{ny}apu}   & sweep \\
    \msa{karang} & \msa{me\alert{ng}arang} & compose \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  The nasal takes the place of the following consonant; voiceless
  p, t, s, k then drop out.  Tagalog does the same
\end{frame}

\begin{frame}{Korean: Rules in Sequence}
  \large
  \begin{tabular}{lll}
    \toprule
    tensing     & \kos{입다} \kor{ip-da}     & [ip̚t͈a] ``wear'' \\
    nasalizing  & \kos{백리} \kor{baek-ri}   & [pɛŋni] ``100 \emph{li}'' \\
    both        & \kos{수업료} \kor{sueop-ryo} & [suʌmnjo] ``tuition'' \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  /r/ becomes /n/ except after a vowel; then a stop becomes a nasal before
  a nasal.  Vowels coalesce in suffixes too: o + i $\to$ oe, a + i $\to$ ae
\end{frame}

\begin{frame}{Japanese: Q and N in Verbs}
  \Large
  \begin{tabular}{lll}
    \toprule
    \jpn{tat-} + \jpn{-ta} & \jpn{ta\alert{t}ta} & stood \\
    \jpn{tor-} + \jpn{-ta} & \jpn{to\alert{t}ta} & took \\
    \jpn{kaw-} + \jpn{-ta} & \jpn{ka\alert{t}ta} & bought \\
    \jpn{hachi} + \jpn{sen} & \jpn{ha\alert{s}sen} & eight thousand \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Q copies the following consonant; kana writes it as small \jps{っ}.
  N assimilates to what follows, and is uvular at the end of a word
\end{frame}

\section{Japanese Pitch Accent}

\begin{frame}{Pitch Accent: Where the Pitch Falls}
  \Large
  \begin{tabular}{llll}
    \toprule
    \jps{雨} & \jpn{a'me}  & HL & rain \\
    \jps{飴} & \jpn{ame}   & LH & candy \\
    \midrule
    \jps{箸} & \jpn{ha'shi} & HL  & chopsticks \\
    \jps{橋} & \jpn{hashi'} & LH(-L) & bridge \\
    \jps{端} & \jpn{hashi}  & LH(-H) & edge \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Tokyo Japanese.  Not a tone on every syllable: each word has at most one
  \txx{accent}, and the pitch falls after it.  Bridge and edge differ only
  on the following particle: \jpn{hashi ga}
\end{frame}

\begin{frame}{Predicting the Pitch}
  \Large
  \begin{tabular}{lll}
    \toprule
    unaccented & \jpn{sakura}   & LHH \\
               & \jpn{shirakaba} & LHHH \\
    accented   & \jpn{za'kuro}  & HLL \\
               & \jpn{ka'makiri} & HLLL \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  The accented mora is H, everything after it L.  Before it, H --- except
  that the first mora is L if the second is H.  Few minimal pairs, and
  Kyoto patterns differ
\end{frame}

\section{Wrap-up}

\begin{frame}{Key Takeaways}
  \Large
  \begin{itemize}
  \item Insular: small, disyllabic; mainland: large, tonal
  \item Sinitic: initials + finals + tone
  \item Korean: three-way stops, many alternations
  \item Japanese: morae, N and Q, pitch accent
  \end{itemize}
  \vfill
  \normalsize
  An allotone depends on its environment; sandhi is a tone changed by
  another tone
\end{frame}

\begin{frame}{Practice}
  \large
  \begin{enumerate}
  \item Malay: add \msa{meN-} to \msa{pakai}, \msa{tanam}, \msa{kirim},
    \msa{beli}
  \item Mandarin: how do you say \zhs{很好} \cmn{hěn hǎo} ``very good''?
  \item Korean: why is \kos{국물} \kor{guk-mul} ``soup'' said [kuŋmul]?
  \item Thai: why does \ths{ข่า} sound low but \ths{ค่า} falling?
  \item Japanese: how many morae in \jpn{Tōkyō}? In \jpn{Nippon}?
  \end{enumerate}
  \vfill
  \normalsize
  Use IPA, tone numbers or H/L as needed
\end{frame}
