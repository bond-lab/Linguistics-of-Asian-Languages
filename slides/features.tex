\section{Word order}

\begin{frame}{Two Word-Order Types Meet in East Asia}
  \large
  \begin{tabular}{llll}
    \toprule
    & Japanese, Korean & Mandarin & Thai, Malay, Viet. \\
    \midrule
    verb, object    & OV           & VO          & VO \\
    adposition      & postposition & preposition & preposition \\
    relative clause & Rel N        & \alert{Rel N} & N Rel \\
    possessor       & Poss N       & Poss N      & N Poss \\
    \bottomrule
  \end{tabular}

  \vfill
  \normalsize
  \begin{itemize}
  \item The features \txx{line up}: OV languages put modifiers before the
    noun, VO languages after it
  \item Mandarin is the odd one out: VO, but relative clauses come first
  \end{itemize}
\end{frame}

\section{Classifiers}

\begin{frame}{Counting Needs a Classifier}
  \large
  \begin{tabular}{llll}
    \toprule
    & & & order \\
    \midrule
    Thai       & \tha{mǎa sǎam \alert{tua}}  & dog three \textsc{clf} & N Num Clf \\
    Malay      & \msa{tiga \alert{ekor} anjing} & three \textsc{clf} dog & Num Clf N \\
    Vietnamese & \vie{ba \alert{con} chó}     & three \textsc{clf} dog & Num Clf N \\
    Mandarin   & \cmn{sān \alert{zhī} gǒu}    & three \textsc{clf} dog & Num Clf N \\
    Japanese   & \jpn{san-\alert{biki} no inu} & three-\textsc{clf} of dog & Num Clf N \\
    \bottomrule
  \end{tabular}

  \vfill
  \normalsize
  \begin{itemize}
  \item Required with numerals and with vague quantifiers like ``many'', ``some''
  \item Number and classifier form a \txx{phrase}, which can stand alone
  \end{itemize}
\end{frame}

\begin{frame}{Why Have Classifiers?}
  \Large
  \begin{itemize}
  \item Nouns name \txx{kinds}, like English \eng{rice} or \eng{water}:
    to count, you need a \txx{unit}
  \item The classifier \txx{individuates}: \eng{three \alert{head} of cattle},
    \eng{two \alert{sheets} of paper}
  \item Classifier languages rarely have \txx{obligatory plural} marking
  \end{itemize}
  \vfill
  \normalsize
  Inventories are shrinking: younger speakers lean on the general
  classifier (Mandarin \cmn{gè}, Japanese \jpn{-tsu})
\end{frame}

\begin{frame}{Thai: The Classifier Phrase Stands Alone}
  \begin{exe}
    \ex
    \gll Mǎa kìi \ul{tua}? \\
         dog how.many \textsc{clf}:animal \\
    \trans ``How many dogs?'' [tha]
    \ex
    \gll (Mǎa) sǎam \ul{tua}. \\
         (dog) three \textsc{clf}:animal \\
    \trans ``Three (dogs).'' [tha]
  \end{exe}
\end{frame}

\begin{frame}{Malay: The Same, in a Different Order}
  \begin{exe}
    \ex
    \gll Berapa \ul{ekor} anjing? \\
         how.many \textsc{clf}:animal dog \\
    \trans ``How many dogs?'' [zsm]
    \ex
    \gll Tiga \ul{ekor} (anjing). \\
         three \textsc{clf}:animal (dog) \\
    \trans ``Three (dogs).'' [zsm]
  \end{exe}
\end{frame}

\begin{frame}{Four Kinds of Counting Word}
  \large
  \begin{tabular}{lll}
    \toprule
    kind & counts & example \\
    \midrule
    \txx{sortal}   & individuals    & \jpn{san-\alert{satsu} no hon} ``three books'' \\
    \txx{mensural} & amounts        & \jpn{ni-\alert{hai} no mizu} ``two cups of water'' \\
    \txx{group}    & collections    & \eng{two \alert{pairs} of shoes} \\
    \txx{repeater} & by the noun itself & Thai \tha{taw} ``vase'' \\
    \bottomrule
  \end{tabular}

  \vfill
  \normalsize
  Sortal classifiers apply across the whole lexicon; the others are limited
\end{frame}

\begin{frame}{Japanese \jpn{-tsu}: The General Classifier}
  \Large
  \begin{tabular}{ll}
    \toprule
    used & example \\
    \midrule
    for the residue        & \jpn{yume-o hito-\ul{tsu} mita} \\
                           & ``I had a dream'' \\
    as a default           & \jpn{hon-o chōdo hito-\ul{tsu} katta} \\
                           & ``I bought exactly one book'' \\
    when the thing is vague & \jpn{nantoka-ga hito-\ul{tsu} hoshii} \\
                           & ``I want one thingamajig'' \\
    \bottomrule
  \end{tabular}

  \vfill
  \normalsize
  Sortal classifiers pick out \txx{kind} or \txx{shape};
  the general one is used when nothing is salient \citep{Bond:Paik:2011}
\end{frame}

\begin{frame}{Measure and Group Words}
  \begin{exe}
    \let\eachwordone=\glossja
    \ex
    \glll 二 杯 の 水 \\
          ni hai no mizu \\
          two \textsc{clf}.cup of water \\
    \trans ``two cups of water'' [jpn]
  \end{exe}

  \vfill
  \Large
  \begin{tabular}{ll}
    \eng{a \alert{flock} of birds} & \eng{a \alert{herd} of cows} \\
    \eng{a \alert{mob} of protestors} & \eng{two \alert{bunches} of bananas} \\
  \end{tabular}
  \vfill
  \normalsize
  Group words are common in English and Thai, rarer in Japanese
\end{frame}

\begin{frame}{Repeaters: The Noun Is Its Own Classifier}
  \begin{exe}
    \ex
    \gll Mii \ul{taw} dòòk-maj cak \ul{taw}? \\
         there.is vase flower how.many vase \\
    \trans ``How many vases are there?'' [tha]
  \end{exe}
  \vfill
  \large
  Common in Thai and Lao, especially when no classifier exists for the noun
\end{frame}

\begin{frame}{Classifiers Can Show Respect}
  \Large
  \begin{tabular}{llll}
    \toprule
    ``three people'' & plain & honorific \\
    \midrule
    Japanese & \jps{三人} \jpn{san-\alert{nin}} & \jps{三名様} \jpn{san-\alert{mei-sama}} \\
    Korean   & \kos{세}~\kos{명} \kor{se \alert{myeong}} & \kos{세}~\kos{분} \kor{se \alert{bun}} \\
    \bottomrule
  \end{tabular}

  \vfill
  \begin{exe}
    \let\eachwordone=\glossja
    \ex
    \glll 何 名 様 です か \\
          nan mei sama desu ka \\
          what \textsc{clf}.person \textsc{hon} is \textsc{q} \\
    \trans ``How many in your party?'' [jpn]
  \end{exe}
  \normalsize
  Khmer has special classifiers for royalty and monks
\end{frame}

\begin{frame}{Classifiers Can Count Indirectly}
  \begin{exe}
    \ex
    \glll 2-\cll{tsū}-no denshimēru \\

\section{Aspect}

\begin{frame}{Aspect, Not Tense}
  \Large
  \begin{tabular}{ll}
    \txx{tense}  & \emph{when} did it happen? \\
    \txx{aspect} & \emph{how} does it unfold? \\
  \end{tabular}
  \vfill
  \begin{itemize}
  \item Many East and Southeast Asian languages have no tense
  \item They mark aspect instead: suffixes, auxiliaries, particles
  \end{itemize}
  \vfill
\end{frame}

\begin{frame}{Kinds of Aspect}
  \Large
  \begin{tabular}{lll}
    \toprule
    \txx{perfective}   & the whole event   & \eng{She ran home} \\
    \midrule
    \multicolumn{3}{l}{\txx{imperfective}: the inside of the event} \\
    \quad progressive  & ongoing           & \eng{She is running} \\
    \quad iterative    & repeated          & \eng{The light is flashing} \\
    \quad durative     & taking time       & \eng{He was sitting} \\
    \quad habitual     & regular           & \eng{She runs every morning} \\
    \bottomrule
  \end{tabular}
\end{frame}

\begin{frame}{Aspect Markers in Mandarin and Cantonese}
  \Large
  \begin{tabular}{lllll}
    \toprule
    & \multicolumn{2}{l}{Mandarin} & \multicolumn{2}{l}{Cantonese} \\
    \midrule
    perfective   & V-\cmn{le}   & \zhs{了} & V-\yue{jó}   & \zht{咗} \\
    progressive  & \cmn{zài}-V  & \zhs{在} & V-\yue{gán}  & \zht{緊} \\
    continuous   & V-\cmn{zhe}  & \zhs{着} & V-\yue{jyuh} & \zht{住} \\
    experiential & V-\cmn{guo}  & \zhs{过} & V-\yue{gwo}  & \zht{過} \\
    delimitative & V-\cmn{yi}-V & \zhs{一} & V-\yue{háh}  & \zht{吓} \\
    habitual     &              &          & V-\yue{hōi}  & \zht{開} \\
    \bottomrule
  \end{tabular}

  \vfill
  \normalsize
  Aspect marking is independent of time reference
\end{frame}

\begin{frame}{Mandarin \cmn{-le}: Completion, Not Past}
  \begin{exe}
    \ex
    \gll Tā kāi-\ul{le} mén, nǐ jiù jìn-qu. \\
         3\textsc{sg} open-\textsc{pfv} door 2\textsc{sg} then enter-go \\
    \trans ``(Once) she opens the door, you go in.'' [cmn]
    \ex
    \gll Nǐ gāo-\ul{le} yìdiǎn. \\
         2\textsc{sg} tall-\textsc{pfv} a.little \\
    \trans ``You've got (a little) taller.'' [cmn]
  \end{exe}
  \vfill
  \large
  \cmn{-le} works for future and conditional events, and with stative verbs
\end{frame}

\begin{frame}{Cantonese: Progressive and Experiential}
  \begin{exe}
    \ex
    \gll Wòhng síujé góng-\ul{gán} dihnwá. \\
         Wong miss talk-\textsc{prog} telephone \\
    \trans ``Miss Wong is talking on the phone.'' [yue]
    \ex
    \gll Léih hái Hēunggóng jyuh-\ul{gwo} géi noih a? \\
         2\textsc{sg} in Hong.Kong live-\textsc{exp} how.long \textsc{sfp} \\
    \trans ``How long did you live in Hong Kong?'' [yue]
  \end{exe}
\end{frame}

\begin{frame}{Cantonese: Habitual \yue{-hōi}}
  \begin{exe}
    \ex
    \gll Kéuih jouh-\ul{hōi} jūngdím ge. \\
         3\textsc{sg} work-\textsc{hab} part.time \textsc{sfp} \\
    \trans ``She normally works part-time.'' [yue]
    \ex
    \gll Kéuih yíhchìhn jā-\ul{hōi} Bēnsí ge. \\
         3\textsc{sg} before drive-\textsc{hab} Benz \textsc{sfp} \\
    \trans ``She used to drive a Mercedes.'' [yue]
  \end{exe}
  \vfill
  \large
  Regular or customary actions, present or past
\end{frame}

\begin{frame}{Cantonese: Delimitative \yue{-háh} ``for a while''}
  \begin{exe}
    \ex
    \gll Tái-\ul{háh}. \\
         look-\textsc{del} \\
    \trans ``Have a look.'' [yue]
    \ex
    \gll Yāusīk-\ul{háh}. \\
         rest-\textsc{del} \\
    \trans ``Have a rest.'' [yue]
  \end{exe}
  \vfill
  \large
  English uses \eng{have a} V: \eng{have a look}, \eng{have a walk}
\end{frame}

\begin{frame}{Malay: Aspect Words Before the Verb}
  \begin{exe}
    \ex
    \gll Saya pun \ul{sudah} beritahu juga semalam. \\
         1\textsc{sg} \textsc{foc} \textsc{pfv} tell also yesterday \\
    \trans ``I already told (you) yesterday.'' [zsm]
    \ex
    \gll Bila saya masuk, dia \ul{sedang} membaca. \\
         when 1\textsc{sg} enter 3\textsc{sg} \textsc{prog} read \\
    \trans ``When I came in, they were reading.'' [zsm]
  \end{exe}
  \vfill
  \large
  \msa{sudah} marks completion, for past \emph{or} future events;
  \msa{sedang} marks ongoing action
\end{frame}

\begin{frame}{Singlish Carries the Pattern into English}
  \large
  \begin{tabular}{lll}
    \toprule
    perfect      & \eng{Eat \alert{already}?}           & ``Have you eaten?'' \\
    experiential & \eng{I eat durian \alert{before}.}   & ``I've had durian before.'' \\
    habitual     & \eng{He \alert{always} late one.}    & ``He's always late.'' \\
    continuative & \eng{Your phone \alert{still} spoil ah?} & ``Still broken?'' \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Singapore Colloquial English uses English words as aspect markers
\end{frame}

\begin{frame}{Summary: Aspect}
  \Large
  \begin{itemize}
  \item Aspect, not tense, is the core time category
  \item Marked by \txx{suffixes} (Sinitic) or \txx{words before the verb} (Malay)
  \item Perfective, progressive, experiential, habitual, delimitative
  \end{itemize}
\end{frame}

\section{Serial Verbs}

\begin{frame}{Serial Verbs: Several Verbs, One Clause}
  \Large
  \begin{itemize}
  \item Two or more verbs, \txx{no conjunction}, one subject
  \end{itemize}
  \vfill
  \begin{tabular}{lll}
    \toprule
    \txx{loose} & a sequence of events & own arguments \\
    \txx{tight} & a single event       & motion, posture \\
    \bottomrule
  \end{tabular}
  \vfill
\end{frame}

\begin{frame}{Loose Serialization: A Sequence of Events}
  \begin{exe}
    \ex
    \gll Dèèng paj³ talaat⁵ sùù⁴ khùang¹ maa² hùan² lèèw⁴. \\
         Deng go market buy stuff come house \textsc{pfv} \\
    \trans ``Deng went to the market, bought some stuff, (and) came home.'' [lao]
  \end{exe}
  \vfill
  \large
  One perfective marker covers the whole sequence
\end{frame}

\begin{frame}{Tight Serialization: One Event}
  \begin{exe}
    \ex
    \gll Dèk wîng pay síi khanǒm. \\
         child run go buy candy \\
    \trans ``The child ran (and) bought candy.'' [tha]
  \end{exe}
  \vfill
  \large
  The verbs share their subject and describe a single unitary event
\end{frame}

\begin{frame}{Culture Decides What Counts as One Event}
  \begin{exe}
    \ex
    \gll nang¹ tii³ lanaat⁴. \\
         sit beat lanaat \\
    \trans ``Play the lanaat sitting.'' [lao]
    \ex[?]{
    \gll nòòn² tii³ lanaat⁴. \\
         lie beat lanaat \\
    \trans ``Play the lanaat lying down.'' [lao]}
  \end{exe}
  \vfill
  \large
  Nobody plays the lanaat lying down --- unless the context makes it sensible
\end{frame}

\begin{frame}{Come and Go Give the Direction}
  \begin{exe}
    \ex
    \gll Khǎw klàp \ul{maa} lɛ́ɛw. \\
         3\textsc{sg} return come \textsc{pfv} \\
    \trans ``She came back already.'' [tha]
    \ex
    \gll Khǎw klàp \ul{pay} lɛ́ɛw. \\
         3\textsc{sg} return go \textsc{pfv} \\
    \trans ``She went back already.'' [tha]
  \end{exe}
  \vfill
  \large
  Directional verbs work like adverbs: ``hither'', ``away''
\end{frame}

\begin{frame}{Three Verbs: Manner, Path, Deixis}
  \begin{exe}
    \ex
    \gll Laaw² ñaang¹ khùn⁵ maa². \\
         3\textsc{sg} walk go.up come \\
    \trans ``She walked up here.'' [lao]
  \end{exe}
  \vfill
  \Large
  \begin{tabular}{lll}
    \tha{ñaang¹} & walk  & manner of motion \\
    \tha{khùn⁵}  & go up & absolute direction \\
    \tha{maa²}   & come  & towards the speaker \\
  \end{tabular}
  \vfill
\end{frame}

\begin{frame}{Verbs That Work as Prepositions}
  \begin{exe}
    \ex
    \gll Laaw² nùng¹ khaw⁵ \ul{haj⁵} khòòj⁵. \\
         3\textsc{sg} steam rice give(=for) 1\textsc{sg} \\
    \trans ``He steamed some rice for me.'' [lao]
    \ex
    \gll Khan² haw² juu¹ \ul{nam²} mèè¹-thaw⁵. \\
         if 1\textsc{pl} live accompany(=with) mother-old \\
    \trans ``If we live with your mother.'' [lao]
  \end{exe}
  \vfill
  \large
  ``give'' marks the beneficiary, ``accompany'' the companion
\end{frame}

\begin{frame}{Vietnamese: ``Arrive'' Means ``About''}
  \begin{exe}
    \ex
    \gll Chúng tôi nhắc \ul{đến} anh luôn. \\
         \textsc{pl} 1\textsc{excl} recall arrive(=about) brother often \\
    \trans ``We often speak about you.'' [vie]
  \end{exe}
  \vfill
  \large
  Serial verbs can mark what is being talked about
\end{frame}

\begin{frame}{Japanese Compound Verbs Form a Cline}
  \large
  \setlength{\tabcolsep}{4pt}
  \begin{tabular}{llll}
    \toprule
    & & literally & means \\
    \midrule
    lexical     & \jps{取り締まる} \jpn{tori-shimaru}  & take-squeeze & manage \\
    \midrule
    aspectual   & \jps{書き上げる} \jpn{kaki-ageru}    & write-raise  & write up \\
                & \jps{歩き出す} \jpn{aruki-dasu}      & walk-out     & start walking \\
    \midrule
    syntactic   & \jps{食べ終わる} \jpn{tabe-owaru}    & eat-finish   & finish eating \\
                & \jps{歩き始める} \jpn{aruki-hajimeru} & walk-begin  & start walking \\
    \bottomrule
  \end{tabular}

  \vfill
  \normalsize
  The aspectual ones are like English particle verbs \citep{Hashimoto:Bond:2005}
\end{frame}

\begin{frame}{Arguments Can Transfer}
  \Large
  \begin{tabular}{l}
    \jps{大阪を食べ歩く} \jpn{Ōsaka-o tabe-aruku} \\
    \quad Osaka-\textsc{acc} eat-walk \\
    \quad ``stroll around Osaka eating'' \\[2ex]
    \jps{寿司を食べ歩く} \jpn{sushi-o tabe-aruku} \\
    \quad sushi-\textsc{acc} eat-walk \\
    \quad ``stroll around eating sushi'' \\
  \end{tabular}
  \vfill
  \normalsize
  The object can belong to either verb
\end{frame}

\begin{frame}{Summary: Serial Verbs}
  \Large
  \begin{itemize}
  \item Several verbs, one subject, no conjunction
  \item \txx{Loose}: a sequence; \txx{tight}: one event
  \item They express direction, beneficiaries, companions --- the work
    prepositions do in English
  \end{itemize}
\end{frame}

\section{Subject and Topic}

\begin{frame}{What Makes an English Subject?}
  \Large
  \begin{tabular}{ll}
    \toprule
    before the verb  & \eng{\ul{She} seems happy.} \\
    its own case     & \eng{I} vs \eng{me}, \eng{they} vs \eng{them} \\
    verb agreement   & \eng{he go\alert{es}} \\
    passive promotes & \eng{\ul{Maxine} was elected.} \\
    any role         & agent, patient, experiencer \\
    \bottomrule
  \end{tabular}

  \vfill
  \normalsize
  European languages add case and person-number agreement;
  the traditional view is that every language has subjects
\end{frame}

\begin{frame}{Topic: What the Sentence Is About}
  \Large
  \begin{itemize}
  \item In English, topic and subject usually coincide:
    \eng{\ul{Cats} eat mice.}
  \item In many Asian languages they \txx{diverge}
  \item There, \txx{topic} may matter more than subject
  \end{itemize}
\end{frame}

\begin{frame}{Thai: Topic First}
  \begin{exe}
    \ex
    \gll Pàakphanang khâaw raw kin ngâay. \\
         Pakphanang rice 1\textsc{pl} eat simple \\
    \trans ``(In) Pakphanang, we ate simply.'' [tha]
  \end{exe}
  \vfill
  \large
  \begin{itemize}
  \item Typically SVO, but OSV and SOV are common in informal speech
  \item Subjects are often left out; order follows the discourse
  \end{itemize}
\end{frame}

\begin{frame}{Mandarin: Topic First}
  \begin{exe}
    \ex
    \gll Zhāngsān wǒ yǐjīng jiàn-guo le. \\
         Zhangsan 1\textsc{sg} already see-\textsc{exp} \textsc{crs} \\
    \trans ``Zhangsan, I've already seen (him).'' [cmn]
  \end{exe}
  \vfill
  \large
  \begin{itemize}
  \item Topics always come first; times and places are often topics:
    \eng{In Taipei, one can eat well}
  \item Subjects are left out when context is clear
  \end{itemize}
\end{frame}

\begin{frame}{Mandarin: Topic \emph{and} Subject}
  \begin{exe}
    \ex
    \gll \ul{Xiàng} bízi cháng. \\
         elephant nose long \\
    \trans ``Elephants, (their) noses are long.'' [cmn]
  \end{exe}
  \vfill
  \large
  The topic sets the frame; the comment has its own subject.
  This is ordinary, not marked
\end{frame}

\begin{frame}{Japanese: \jpn{wa} Marks Topic, \jpn{ga} Subject}
  \begin{exe}
    \ex
    \gll Hi \ul{wa} nobor-u. \\
         sun \textsc{top} rise-\textsc{prs} \\
    \trans ``The sun rises.'' (general) [jpn]
    \ex
    \gll Hi \ul{ga} nobor-u. \\
         sun \textsc{nom} rise-\textsc{prs} \\
    \trans ``The sun is rising.'' (this one) [jpn]
  \end{exe}
  \vfill
  \large
  \jpn{wa}: generic or contrastive, and for times and places;
  \jpn{ga}: the event as a whole
\end{frame}

\begin{frame}{Tagalog: A Trigger System}
  \begin{exe}
    \ex
    \gll Mag-aalis ang bata ng laruan sa kahon. \\
         \textsc{at}-will.take.out \textsc{trig} child \textsc{pat} toy \textsc{dir} box \\
    \trans ``The child will take a toy out of the box.'' [tgl]
  \end{exe}
  \vfill
  \large
  The verb's affix says which role the \txx{trigger} (\msa{ang}) NP has;
  the trigger is definite --- neither subject nor topic quite fits
\end{frame}

\begin{frame}{Acehnese: Actor and Undergoer}
  \begin{exe}
    \ex
    \gll \ul{Geu}-jak. \\
         3-go \\
    \trans ``(S)he goes.'' [ace]
    \ex
    \gll Lôn rhët-\ul{lôn}. \\
         1\textsc{sg} fall-1\textsc{sg} \\
    \trans ``I fall.'' [ace]
  \end{exe}
  \vfill
  \large
  Prefixes mark \txx{actors}, suffixes mark \txx{undergoers} --- not
  subject and object
\end{frame}

\begin{frame}{Summary: Subject and Topic}
  \Large
  \begin{itemize}
  \item European languages are \txx{subject}-prominent;
    East Asian languages \txx{topic}-prominent
  \item Topic and subject need not coincide
  \item Tagalog and Acehnese question whether subjects are universal
  \end{itemize}
\end{frame}

\section{Sentence Final Particles (SFP)}

\begin{frame}{Particles Carry the Speaker's Attitude}
  \Large
  \begin{itemize}
  \item At the \txx{end} of the sentence
  \item Speech acts: asking, requesting, persuading
  \item Feelings: surprise, doubt, hesitation
  \end{itemize}
  \vfill
  \begin{tabular}{lll}
    \toprule
    & one every & particles \\
    \midrule
    Cantonese & \alert{1.5} seconds & 30+, in 100+ combinations \\
    Mandarin  & \alert{6} seconds   & about 7 common ones \\
    \bottomrule
  \end{tabular}
  \vfill
  \normalsize
  Like English intonation, but explicit. Common in Thai, Lao and
  Vietnamese too; rarer in formal and written language
\end{frame}

\begin{frame}{Cantonese: Particles in Questions}
  \begin{exe}
    \ex
    \gll Léih gú gam yùhngyih \ul{áh}? \\
         2\textsc{sg} guess so easy \textsc{sfp} \\
    \trans ``You think it's that easy, do you?'' [yue]
  \end{exe}
  \vfill
  \Large
  \begin{tabular}{ll}
    \yue{áh} & surprise or disapproval \\
    \yue{há} & presupposing agreement \\
    \yue{mē} & surprise, often rhetorical \\
  \end{tabular}
  \vfill
\end{frame}

\begin{frame}{Cantonese: Directive Particles}
  \begin{exe}
    \ex
    \gll Taai chòuh \ul{la}, ngóh fan m̀h dóu. \\
         too noisy \textsc{sfp} 1\textsc{sg} sleep not able \\
    \trans ``It's too noisy, I can't sleep.'' [yue]
    \ex
    \gll Gám jauh dāk \ul{lo}. \\
         so then okay \textsc{sfp} \\
    \trans ``That'll be all right, won't it?'' [yue]
  \end{exe}
  \vfill
  \large
  \yue{la}: current relevance; \yue{lo}: invites agreement
\end{frame}

\begin{frame}{Conclusions}
  \Large
  \begin{itemize}
  \item \txx{Classifiers} are needed to count, and sort nouns by kind,
    shape and status
  \item \txx{Aspect}, not tense, organizes time
  \item \txx{Serial verbs} do the work of prepositions and adverbs
  \item \txx{Topic} can matter more than subject
  \item \txx{Particles} carry the speaker's attitude
  \end{itemize}
  \vfill
  \normalsize
  Southeast Asian languages package meaning differently from European ones
\end{frame}
