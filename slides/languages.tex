\begin{frame}{The major language families of SEA}
  \begin{itemize}

\item \textbf{Austronesian}: coastal Vietnam, Burma and western peninsular SEA

\item \textbf{Austroasiatic} (Mon-Khmer; the Munda branch is only in South
  Asia): scattered throughout MSEA, reaching NE India (Khasian branch)

\item \textbf{Sino-Tibetan}
\begin{itemize}
\item \textbf{Tibeto-Burman}: Burma, NE India, the mountains of western and
  northern Thailand, Laos and northern Vietnam
\item \textbf{Sinitic}: mostly China
\end{itemize}

\item \textbf{Tai-Kadai}: southern China, SEA, NE India

\item \textbf{Hmong-Mien} (Miao-Yao): mostly southern China, extending into SEA
\item \textbf{Korean, Japanese and Ainu}: Korean Peninsula and Japan
\end{itemize}  
Genetically unrelated, yet sharing many features: MSEA is a linguistic area

\end{frame}

\begin{frame}{Human and linguistic diversity in MSEA}
\begin{itemize}

\item Mainland Southeast Asia (MSEA) has high human diversity compared to
  the global average
  \begin{itemize}
  \item In social structure, material culture and genetic markers
  \end{itemize}
\item Linguistic diversity, however, is regarded as relatively low, although
  there are around 2,000 languages. Why?
\end{itemize}

\begin{frame}
  \begin{itemize}
    \item \textbf{Language families}: most languages belong to a few families
      (Austroasiatic, Tai-Kadai, Sino-Tibetan, Hmong-Mien)
    \item \textbf{Contact and convergence}: trade, migration and exchange
      have made unrelated languages alike
    \item \textbf{Multilingualism and shift}: lingua francas (Thai,
      Vietnamese) overshadow the smaller languages
    \item \textbf{Classification}: where linguists draw the line between
      language and dialect changes the count
\end{itemize}
\end{frame}

\begin{frame}{Mon-Khmer branch of Austroasiatic}
\begin{itemize}
\item Aslian (mostly endangered): southern Thailand and western Malaysia

\begin{frame}
  According to \citet[p.~38]{goddard:2005}
  \begin{itemize}
  \item In the Chinese tradition any variety other than Mandarin is a \emph{dialect}
  \item This reflects the cultural and historical unity of China and the use of a common script
    \\ although not all words can be written in Hanzi [FCB]
\item In normal usage, dialects are mutually intelligible
\item  Linguists usually refer to the seven language groupings  as \emph{Sinitic languages}
\item But  some of these languages also have many mutually unintelligible varieties
\item E.g.,  Min has as many as  nine mutually unintelligible varieties in the Fujian province alone
(\citealt{li:1992}, cited in \citealt[p.~38]{goddard:2005})
\item There are probably hundreds of mutually unintelligible  Sinitic
languages in China
\end{itemize}

\begin{frame}{Sinospheric features}

Sino-Tibetan languages in SEA tend to share features with Sinitic, e.g.

\begin{itemize}
\item complex tone
\item classifiers
\item limited morphology
\item isolating word formation
\item pragmatically determined syntax
\item reliance on zero anaphora
\end{itemize}
\end{frame}

\begin{frame}{Indospheric features}


Sino-Tibetan languages in South Asia tend to share features with Indic, e.g.

\begin{itemize}
\item simple tone systems or no lexical tone
\item morphologically complex stems
\item fusional or agglutinative word formation
\item breathy and retroflex consonants
\item pronominal cross-referencing on the verb
\item well-developed case-marking paradigms
\end{itemize}
\end{frame}

\section{Mainland Southeast Asia as a linguistic area}

\begin{frame}{MSEA as a Linguistic Area}
    \begin{itemize}
        \item Languages of different families converge when in close contact
        \item A \txx{linguistic area} (\textit{Sprachbund}): unrelated
          languages sharing features
        \item MSEA is one, after centuries of mutual influence
        \item Shared features in phonology, lexicon and grammar
          (e.g. limited syllable structures, lexical tone)
        \item Many features shared with non-mainland languages too
    \end{itemize}
\end{frame}


% Slide 2
\begin{frame}{Shared Phonological Features}
    \begin{itemize}
        \item A limited range of syllable structures
        \item Lexical tone across the region: Vietnamese, Thai, most
          Chinese dialects
        \item Mon-Khmer languages mostly lack tone (with exceptions)
        \item Convergence from prolonged contact
    \end{itemize}
\end{frame}

\begin{frame}{Shared Syntactic Features}
    \begin{itemize}
        \item Serial verb constructions
        \item Topic prominence: word order follows the topic
        \item Ellipsis: context supplies what is left out
        \item Sentence-final particles express the speaker's attitude
    \end{itemize}
\end{frame}
\begin{frame}{Topic Prominence: Word Order Examples}
    \begin{itemize}
        \item \textbf{Mandarin Chinese}:
            \begin{exe}
                \ex Topic-comment structure: \\
                \glll 这 本 书 ， 我 看 过 了 。 \\
                 zhè běn shū , wǒ kàn guò le . \\
                 this \textsc{clf} book {} 1\textsc{sg} read \textsc{exp} \textsc{crs} {} \\
                \glt ``This book, I have read.'' [cmn]
            \end{exe}
        \item \textbf{Japanese}:
          \begin{exe}
            \let\eachwordone=\glossja
            \ex Topic marker \jps{は} \jpn{wa}: \\
            \glll 象 は 鼻 が 長い 。 \\
            zō wa hana ga nagai . \\
            elephant \textsc{top} nose \textsc{nom} long {} \\
            \glt ``Elephants have long noses.'' [jpn]
          \end{exe}
    \end{itemize}
\end{frame}

% Slide 6
\begin{frame}{Topic Prominence and Ellipsis}
    \begin{itemize}
        \item Sentence structure follows the discourse
        \item Less rigid word and phrase order
        \item Ellipsis: understood participants are left out
          (e.g. the subject, when clear from context)
        \item Context does much of the work
    \end{itemize}
\end{frame}
\begin{frame}{Ellipsis: Omitting Understood Participants}
  \begin{exe}
    \ex \textbf{Japanese}:
    \gll  $\phi$ 行き ます 。  \\
    $\phi$ iki masu . \\
    \glt  ``(I) am going.'' [jpn]

   \ex \textbf{Mandarin Chinese}:
                 \gll  $\phi$ 吃 了 吗？ \\
                 $\phi$ chī le ma? \\
                 \glt``Have (you) eaten?'' [cmn]
         \ex \textbf{Thai}:
                 \glll  $\phi$ \ths{ไป} \ths{ไหน}? \\
                 $\phi$ pay nǎy? \\
                 $\phi$ go where? \\
                 \glt ``(You) go where?'' [tha]
     \end{exe}

  \end{frame}

  % Slide 7
\begin{frame}{Sentence-Final Particles}
    \begin{itemize}
        \item Small expressive words at the end of a sentence
        \item Mark the speaker's feelings or attitude, or a question
        \item Thai \ths{นะ} \tha{ná} softens a statement
        \item Common across mainland Southeast Asia
    \end{itemize}
  \end{frame}
  
\begin{frame}{Sentence-Final Particles: Examples}
  \begin{itemize}
  \item \textbf{Thai}:
    \begin{itemize}
    \item \ths{คุณสบายดีไหมคะ}? \\
      (\tha{khun sabaay dii mǎy khá}?) \\
      ``How are you?'' (\tha{khá} adds politeness from a female speaker)
    \item \ths{ไปนะ} \\
      (\tha{pay ná}) \\
      'I'm leaving now, okay?' (\tha{ná} softens the statement)
    \end{itemize}
  \item \textbf{Mandarin Chinese}:
    \begin{exe}
      \ex \gll 我们 走 吧 。\\
      wǒmen zǒu ba .\\
      we go SUGG .\\
      \trans `Let's go.' (\textit{ba} indicates a suggestion)
      
      \ex \gll 好 吃 吗 ？\\
      hǎo chī ma ? \\
      good eat Q ? \\
      \trans `Is it delicious?' (\textit{ma} turns a statement into a question)
    \end{exe}
  \end{itemize}
\end{frame}

\begin{frame}{Summary of Areal Features}
    \begin{itemize}
        \item Shared features: lexical tone, classifiers, \ldots
        \item But variation within families
          \begin{itemize}
          \item Mon-Khmer: some languages lack tone
          \item Tibeto-Burman: mostly verb-final, with postpositions
          \item Sinitic: both prepositions and postpositions
          \end{itemize}
    \end{itemize}
\end{frame}

\section{Geography, history and language policy}

% Slide 9
\begin{frame}{Rivers and mountains shape language distribution}
    \begin{itemize}
        \item Major rivers: Irrawaddy, Chao Phraya, Mekong, Red River
        \item Fertile deltas: large populations, languages spread
        \item Mountain ranges: linguistic boundaries
        \item Southern China belongs to the same linguistic area
    \end{itemize}
\end{frame}

% Slide 10
\begin{frame}{Uplands and Lowlands}
    \begin{itemize}
        \item Lowlands: wet rice supports large populations (Thai, Vietnamese)
        \item Uplands: shifting agriculture supports smaller groups
          (Hmong villages at 1,000–1,500 m)
        \item Elevation sets how people farm and settle
    \end{itemize}
\end{frame}

% Slide 11
\begin{frame}{Historical Language Dispersal}
    \begin{itemize}
        \item Mon-Khmer speakers inland by 4,000 years ago
        \item Tai speakers came down from southern China 2,000 years ago,
          following the river valleys into the lowlands
        \item Han Chinese expanded southward, driving language shift
        \item Austronesian in southern Vietnam (the Cham empire)
    \end{itemize}
\end{frame}

% Slide 12
\begin{frame}{Rise and Fall of Empires}
    \begin{itemize}
    \item 1,000 years ago: Mon-Khmer empires dominated
      (e.g. the Khmer empire of Angkor)
    \item Tai kingdoms grew in Laos and Thailand, under Khmer cultural influence
    \item Mon-Khmer groups shifted to Tai languages
    \item Tibeto-Burman peoples moved into Myanmar, ending Mon dominance
    \item Mon now survives in small pockets of eastern Burma
    \end{itemize}
  \end{frame}

% Slide 13
\begin{frame}{Warfare and Migration}
    \begin{itemize}
        \item Wars between Thai, Burmese and Vietnamese reshaped demographics
        \item Large armies with elephant corps: the kingdom of Lan Xang
          (``million elephants'')
        \item Migration shifted the ethnic and linguistic mix
        \item Cultural exchange through conquest and alliance
    \end{itemize}
\end{frame}

% Slide 14
\begin{frame}{Hmong-Mien Peoples}
    \begin{itemize}
        \item In southwestern China until about 150 years ago, under Han domination
        \item Moved south under pressure and conflict
        \item High mountain villages (1,000--1,500 m)
        \item Minority status preserved culture and language
     \end{itemize}
\end{frame}

% Slide 15
\begin{frame}{Colonial Rule in the 19th Century}
    \begin{itemize}
        \item British: Myanmar and Malaya
        \item French: Indochina (Vietnam, Laos, Cambodia)
        \item Thailand stayed independent, but lost some territory
        \item Colonial borders became the modern nation-states
    \end{itemize}
\end{frame}

\begin{frame}[allowframebreaks]{Official languages and multilingualism}

  \begin{description}
  \item[Singapore] Four official languages: English, Malay, Mandarin and
    Tamil. Malay is the \textbf{national} language and the anthem is sung in
    it, but English is the language of administration and the medium of
    education; every child also studies an assigned ``mother tongue''.
  \item[Indonesia] \textit{Bahasa Indonesia} is the sole national language,
    standardised from Riau Malay and adopted by the Youth Pledge of 1928.
    It is the first language of only a minority, and that is rather the
    point --- it belonged to no dominant group. Javanese has far more
    native speakers (about 80 million) and no official status. Some 700
    local languages.
  \item[Philippines] Filipino (based on Tagalog) and English are official;
    the roughly 180 other languages are ``auxiliary official languages in
    the regions''. Mother-tongue-based multilingual education was brought
    into the early grades in 2012 and remains contested.
  \item[China] The 2000 Law on the Standard Spoken and Written Chinese
    Language makes Putonghua and simplified characters the standard for
    government, education and broadcasting. Minority languages have
    constitutional protection; Sinitic ``dialects'' such as Cantonese and
    Hokkien have none.
  \item[Timor-Leste] Tetum and Portuguese are official, Indonesian and
    English are ``working languages''. Around 30 local languages, mostly
    Austronesian with a handful of Papuan ones.
  \end{description}

  \bigskip

  Official status, national identity, and what people actually speak at
  home are three different things.

\end{frame}

\begin{frame}{Conclusion: Linguistic Convergence}
    \begin{itemize}
        \item Mainland Southeast Asia is a linguistic area
        \item Shared features come from long contact and mutual influence
        \item Geography and history decided where the languages are
    \end{itemize}
\end{frame}

\section{Summary}

\begin{frame}{Summary I}

  \begin{itemize}
  \item Five major language families in Mainland SEA
  \item Many languages, but relatively low linguistic diversity
  \item Probably due to centuries of language contact and population movement
  \end{itemize}
\end{frame}


\begin{frame}{Summary II}

MSEA languages are characterized by:

\begin{itemize}
\item isolating word formation\\
\item tone\\
\item absence of inflectional morphology\\
\item classifier systems\\
\item aspectual and mood systems rather than tense systems\\
\item sentence final particles\\
\item extensive zero anaphora\\
\item serial verb constructions
\item re-duplication
\end{itemize}

Although there are many exceptions

\end{frame}
